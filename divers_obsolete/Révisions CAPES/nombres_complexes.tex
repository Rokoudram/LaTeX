\section{Généralités}

\subsection{}
On pose $z_1=1+i\tan(\theta)$ et $z_2=1+\frac{i}{\tan(\theta)}$\\
Déterminer pour quelles valeurs de $\theta$ les nombres $z_1$ et $z_2$ existent, puis donner leur module et un de leurs arguments.

\subsection{}
Soient $a,b,c\in\UU$

Démontrer que $\abs{a+b+c}=\abs{\frac{1}{a}+\frac{1}{b}+\frac{1}{c}}=\abs{ab+bc+ca}$

\subsection{}
\begin{enumerate}
	\item Soit $z\in\UU\setminus\{1\}$. \quad
	Montrer que $\Reel\left(\frac{1}{1-z}\right)=\frac{1}{2}$
	\item Soit $z\in\UU\setminus\{1\}$.\quad
	Montrer que $\frac{z+1}{z-1}\in i\RR \quad \forall z \neq 1$
	\item Soient $z,w\in\UU$. \quad
	Montrer que $\frac{(z+w)^2}{zw} \in \RR_+$
	\item Soient $z,w\in\UU$ tels que $zw\neq-1$. \quad
	Montrer que $\frac{z+w}{1+zw} \in \RR$ 
\end{enumerate}

\subsection{}
Pour $a\in\CC\setminus\UU$, on considère la fonction $f_a:z\mapsto\dfrac{z-a}{1-\barre{a}z}$
\begin{enumerate}
	\item Montrer que $f_a$ est définie sur $\UU$ puis que $f_a\left(\UU\right)=\UU$
	\item Montrer que $f_a$ définit une bijection de $\UU$ dans $\UU$
\end{enumerate}

\subsection{}

Calculer les sommes $A_n=\somme{k\in\NN}{}x^k\cos(\theta k)$ \quad et \quad $B_n(x)=\somme{k\in\NN}{}x^k\sin(\theta k)$

\subsection{}
Soient $a,b,c\in\UU$

Montrer que $a+b+c \in \UU \Longleftrightarrow (a+b)(b+c)(c+a)=0$

\section{Géométrie du plan complexe}

\subsection{}
Soient $A$, $B$, $C$ et $D$ des points du plan complexe d'affixes respectives $a$, $b$, $c$ et $d$.
\begin{enumerate}
	\item Si $a+c=b+d$, que peut-on conclure sur le quadrilatère $ABCD$ ?
	\item Si $A$ et $C$ sont deux sommets opposés d'un carré, exprimer $b$ et $d$ en fonction de $a$ et $c$
\end{enumerate}

\subsection{}

\subsection{}
