\textit{Pour chaque \textbf{Exercice}, plusieurs affirmations sont proposées. Pour chaque affirmation, vous direz si elle est vraie ou fausse en cochant la réponse choisie sur la feuille de réponses. \\
Aucune justification n’est demandée. \\
Une réponse fausse sera pénalisée par des points négatifs. \\
Pour chaque exercice, le total des points obtenu ne peut être strictement négatif. \\
Aucun point n’est enlevé en l’absence de réponse. \\
Les exercices sont tous indépendants.}

\section*{Première partie - Calculs}

\subsection{Vrac 1}
\begin{enumerate}
    \item Pour tout nombre réel $x$ , $\ln\left(\ln\left(e^{e^x}\right)\right) = x$
    \item $\ln\left(\frac{\sqrt{5}+1}{2}\right)+\ln\left(\frac{\sqrt{5}-1}{2}\right)=0$
    \item Pour tous nombres réels $x$ et $y$ strictement positifs, $x^{\ln(y)} = y^{\ln(x)}$
    \item $\sqrt{\frac{3}{5}+\sqrt{\sqrt{\frac{5}{2}+\sqrt{\frac{9}{4}}}-\frac{1}{25}}}=2$
\end{enumerate}



\subsection{Vrac 2}
\begin{enumerate}
	\item Le nombre $\sqrt{8}+\dfrac{1}{3+2\sqrt{2}}$ est un entier positif.
	\item L'équation $\ln(x+1)+\ln(x+3)=\ln(x+7)$ admet deux solutions réelles.
	\item Pour tout nombre réel $x$ strictement positif, $\dfrac{1}{1+\frac{1}{1+\frac{1}{x}}}=\dfrac{x+2}{2x+1}$
	\item Le nombre entier $4^5\times20^3\times25^{12}\times8^4$ se termine par $27$ zéros.
	\item Soit $a$ un nombre réel. Le nombre $\left(2\sqrt{3}-3\sqrt{2}\right)^2-a\sqrt{6}$ est entier si et seulement si $a=12$
\end{enumerate}



\subsection{Vrac 3}
\begin{enumerate}
	\item Pour tout nombre réel $x$ , $\dfrac{e^{4x}-1}{e^{2x}+1}=(e^x-1)(e^x+1)$
	\item $\ln(1000)-\ln(400)-\ln(40)=2\ln\left(\frac{1}{4}\right)$
	\item  $\dfrac{4^{10}-4^9}{2^{16}-2^{14}}=16$
	\item L'équation $e^{2x}-6e^x+5=0$ admet deux solutions réelles.
\end{enumerate}



\subsection{Second degré}
Soit $m$ un nombre réel quelconque.
\begin{enumerate}
	\item L'équation $x+\frac{1}{x}=\frac{1}{m}$ admet une unique solution si et seulement si $m=\frac{\sqrt{2}}{2}$
	\item L'équation $x^2-(m+1)x-m$ admet toujours deux solutions réelles.
\end{enumerate}



\subsection{Vrac 4}
\begin{enumerate}
    \item $\dfrac{\sqrt{e^3\times e^{-7}}}{2e}\times\sqrt{16e^6}=2$
    \item $e^{2\ln(3)}-e^{3\ln(2)}=1$
    \item Pour tout nombre réel $x$ différent de $0$ et $-1$, $\dfrac{x^2}{1+x}+\dfrac{x}{\frac{1}{x}+\frac{1}{x^2}}=x^2$
    \item Pour tout nombre réel $\lambda$ strictement supérieur à $1$, $\left(\dfrac{\lambda+\sqrt{\lambda}}{\lambda-\sqrt{\lambda}}\right)^2=\lambda^2-\lambda$
\end{enumerate}



\subsection{Vrac 5}
\begin{enumerate}
	\item $\sqrt{17+12\sqrt{2}}=5+\sqrt{8}$
	\item Pour tout nombre réel $x$ différent de $-1$ et $3$ , $\dfrac{1}{x-3}-\dfrac{4}{x^2-2x-3}=\dfrac{1}{x+1}$
	\item Pour tout nombre réel $x$ , $1-2\ln(x^2)=\ln\left(\frac{e}{x^3}\right)-\ln(x)$
	\item Si trois nombres réels non nuls $a$, $b$ et $c$ sont tels que $\frac{1}{a}+\frac{1}{b}+\frac{1}{c}=0$ , alors  \\$(a+b+c)^2=a^2+b^2+c^2$
	\item Si $x$ et $y$ sont deux nombres réels non nuls opposés, alors l'inverse de la moyenne des inverses des carrés de $x$ et $y$ vaut $\abs{xy}$
\end{enumerate}



\subsection{Relations entre deux réels}
On considère $x$ et $y$ deux nombres réels non nuls.
\begin{enumerate}
    \item Si $\frac{1}{x}+\frac{2}{y}=0$ , alors $xy<0$
    \item Si $\frac{x}{y}+\frac{y}{x}=1$ , alors $(x-y)^2=0$
    \item Si $xy>0$ , alors $x^2+y^2>xy$
\end{enumerate}

	

\section*{Deuxième partie - Fonctions}

\subsection{Relations fonctionnelles}
\begin{enumerate}
	\item Soit $f$ une fonction vérifiant $f\left(x+\frac{1}{x}\right)=x^2+\frac{1}{x^2}$ pour tout nombre réel $x$ non nul.\\
	On peut affirmer que $f(2)=2$
	\item Soit $f$ une fonction affine telle que $f(1)=2f(3)$. On peut affirmer que $f(5)=0$
\end{enumerate}



\subsection{Logarithme composé}
On considère la fonction $f$ définie sur $\RR$ par $f(x)=\ln(1+x^2)$. On note $\Ccal_f$ sa courbe représentative dans un repère orthonormé et $\Tcal$ sa tangente au point d'abscisse $1$
\begin{enumerate}
	\item La fonction $f$ est positive sur $\RR$
	\item La fonction $f$ est convexe sur $[1~;~+\infty[$
	\item Une équation de $\Tcal$ est $y=x+\ln\left(\frac{2}{e}\right)$
	\item $\Tcal$ se situe au-dessus de $\Ccal_f$ sur $\RR$
	\item L'axe des abscisses est une tangente à $\Ccal_f$
\end{enumerate}



\subsection{Étude d'une fonction}
On définit la fonction $g$ par $g(x)=x+\ln(x^2)$ pour tout $x$ non nul. On note $\Ccal_g$ sa courbe représentative dans un repère orthonormé.
\begin{enumerate}
	\item $\Ccal_g$ admet exactement une asymptote horizontale.
	\item $\Ccal_g$ admet exactement une tangente horizontale.
	\item $g''(x)=\frac{-2}{x^2}$
	\item $g$ admet un maximum sur $]-\infty~;~0[$ qui vaut $2\ln\left(\frac{2}{e}\right)$
\end{enumerate}


\subsection{Cotangente hyperbolique}
Soit $f$ la fonction définie sur $\RR^*$ par $f(x)=\dfrac{e^x+1}{e^x-1}$\\
On note $\Ccal_f$ sa courbe représentative dans un repère orthonormé.
\begin{enumerate}
    \item L'axe des abscisses n'est pas une asymptote à $\Ccal_f$
    \item L'axe des ordonnées n'est pas une asymptote à $\Ccal_f$
    \item Pour tout nombre réel $x$ non nul, $f(-x)=-f(x)$
    \item La fonction $f$ est décroissante sur $]-\infty~;~0[$
    \item L'équation $f(x)=1$ admet au moins une solution sur $]0~;~+\infty[$
\end{enumerate}



\subsection{Tangente hyperbolique}
On définit sur $\RR$ la fonction $f$ par $f(x)=\frac{e^x-e^{-x}}{e^x+e^{-x}}$
\begin{enumerate}
	\item Pour tout $x$ réel , $f(x)=\frac{(e^x+1)(e^x-1)}{e^{2x}+1}$
	\item $f$ est majorée par $1$ sur $\RR$
	\item La courbe représentative de la fonction $f$ admet exactement une asymptote horizontale.
	\item $\limx f'(x) = 0$
\end{enumerate}



\subsection{T.V.I.}
On considère la fonction $f$ définie par $f(x)=\sqrt{x}(x-3)$ pour tout $x$ positif ou nul.
\begin{enumerate}
	\item Pour tout $x$ strictement positif, $f'(x)=\frac{2x-3}{2\sqrt{x}}$
	\item La fonction $f$ admet un minimum  sur $[0~;~+\infty[$ égal à $-2$ 
	\item L'équation $f(x)=a$ admet exactement une solution pour tout nombre réel $a$ positif.
	\item La tangente à la courbe représentative de la fonction $f$ au point d'abscisse $4$ coupe l'axe des abscisses au point de coordonnées $\left(\frac{29}{9};0\right)$
\end{enumerate}



\subsection{Double T.V.I.}
Soient $f$ et $g$ deux fonctions continues strictement positives et strictement croissantes définies sur un intervalle $[a~;~b]$\\
On suppose que $2$ admet un antécédent par $f$ et que $4$ admet un antécédent par $g$
\begin{enumerate}
    \item L'équation $f(x)+g(x)=6$ admet exactement une solution sur $[a~;b~]$
    \item L'équation $f(x)-g(x)=-2$ admet au moins une solution sur $[a~;~b]$
    \item L'équation $\frac{f(x)}{g(x)}=\frac{1}{2}$ admet au moins une solution sur $[a~;~b]$
\end{enumerate}



\subsection{Polynôme de degré 3}
On définit la fonction $f$ par $f(x)=x^3-12x+6$ pour tout nombre réel $x$.  On note $\Ccal_f$ sa courbe représentative dans un repère orthonormé.
\begin{enumerate}
	\item $\Ccal_f$ admet un point d'inflexion situé sur l'axe des ordonnées.
	\item $f$ est croissante sur $]-\infty~;~-3]$
	\item L'équation $f(x)=11$ admet exactement une solution sur l'intervalle $[-2;3]$
\end{enumerate}



\subsection{Valeurs absolues}
Soit $f$ une fonction définie sur $\RR$ et vérifiant $\abs{f(x)}\geq f(\abs{x})$ pour tout nombre réel $x$
\begin{enumerate}
	\item $f$ est négative sur $[0~;~+\infty[$
	\item $f$ est positive sur $]-\infty~;~0]$
	\item $f(x)\neq0$ pour tout nombre réel $x$
\end{enumerate}



\subsection{Ensemble de fonctions}
On note $E$ l'ensemble des fonctions définies sur $\RR$ vérifiant $\abs{f(x)}\geq x$ pour tout nombre $x$ réel.
\begin{enumerate}
	\item Aucune fonction constante n'appartient à $E$
	\item La fonction carrée appartient à $E$
	\item La fonction logarithme népérien n'appartient pas à $E$
	\item Aucune fonction de $E$ n'a une courbe représentative admettant une asymptote horizontale.
\end{enumerate}



\subsection{Minoration}
Soit $f$ une fonction définie sur $\RR$ vérifiant $\abs{f(x)-4}\leq e^{-x}$ pour tout nombre réel $x$.\\
On note $\Ccal_f$ sa courbe représentative dans un repère orthonormé.
\begin{enumerate}
	\item $\Ccal_f$ n'admet pas d'asymptote horizontale.
	\item $\Ccal_f$ n'admet pas d'asymptote verticale.
	\item $\limx f(x)=4$
	\item $\limite{x\to-\infty} f(x)=+\infty$	
\end{enumerate}



\subsection{Équation différentielle}
On considère $f$ une fonction dérivable définie sur $\RR$ vérifiant $f(x)=6-2f'(x)$ pour tout nombre réel $x$ ainsi que $f(2)=2$
\begin{enumerate}
    \item On peut en déduire que $f(2)=f'(2)$
    \item Une équation de la tangente à la courbe représentative de la fonction $f$ dans un repère orthonormé est $y=2x-2$
    \item  $\limite{x\to-\infty} f(x)=-\infty$
\end{enumerate}



\subsection{Primitives}
\begin{enumerate}
	\item Il existe au moins une fonction définie sur $\RR$ dont toutes les primitives sont positives sur $\RR$
	\item Soit $F$ l'unique primitive de la fonction $f$ définie sur $\RR$ par $f(x) = x^5+x^2+x+1$ s'annulant en $1$. L'image de $-1$ par $F$ est positive.
	\item On considère $a$ et $b$ deux nombres réels quelconques.\\
	Si la fonction $F$ définie sur $\RR$ par $F(x) = (ax+b)e^{2x}$ est une primitive de la fonction $f$ définie par $f(x) = xe^{2x}$ , alors $\frac{-b}{a}=2$
\end{enumerate}



\subsection{Convexité}
\begin{enumerate}
	\item Si la fonction $f$ est convexe sur $\RR$ alors la fonction $g$ définie par $g(x)=e^{f(x)}$ est convexe sur $\RR$
	\item Soit $f$ la fonction définie sur $\RR_+$ par $f(x) = x^3-x^2-x+1$. La courbe représentative de $f$ admet exactement deux points d'inflexion.
	\item Soient $f$ une fonction définie et deux fois dérivable sur un intervalle $[a~;~b]$ et $\Ccal_f$ sa courbe représentative dans un repère orthonormé. On note $A$ le point de $\Ccal_f$ d'abscisse $a$ et $B$ le point de $\Ccal_f$ d'abscisse $b$. Si $f'$ est décroissante sur $[a~;~b]$ toute tangente à $\Ccal_f$ se situe en-dessous du segment $[AB]$
\end{enumerate}



\subsection{Polynôme et exponentielle}
Soit $f$ la fonction définie sur $\RR$ par $f(x)=x^2+x+e^{-x}$
\begin{enumerate}
	\item La courbe représentative de la fonction $f$ se situe au-dessus de la courbe représentative de la fonction $f''$ sur l'intervalle $[-2~;~1]$
	\item L'équation $f'(x)=0$ admet une unique solution sur $\RR$
	\item $f$ admet un minimum sur $\RR$ et ce minimum est atteint en $x=0$
\end{enumerate}



\section*{Troisième partie - Suites numériques}

\subsection{Suites géométriques}
On considère $(u_n)_{\nN}$ une suite géométrique de raison $2$ et $(v_n)_{\nN}$ une suite géométrique de raison $3$.\\
On suppose les suites $(u_n)_{\nN}$ et $(v_n)_{\nN}$ non nulles.
\begin{enumerate}
	\item La suite $(u_n\times v_n)_{\nN}$ est géométrique.
	\item La suite $(u_n)_{\nN}$ diverge vers $+\infty$
	\item La suite $(v_n)_{\nN}$ est croissante.
	\item La suite $\left(\frac{u_n}{v_n}\right)_{\nN}$ converge.
\end{enumerate}



\subsection{Suites arithmétiques}
On considère $(u_n)_{\nN}$ une suite arithmétique de raison $3$ et $(v_n)_{\nN}$ une suite arithmétique de raison $2$.
\begin{enumerate}
	\item La suite $(u_n - v_n)_{\nN}$ est arithmétique.
	\item La suite $(u_n)_{\nN}$ diverge vers $+\infty$
	\item La suite $(v_n)_{\nN}$ est croissante.
\end{enumerate}



\subsection{Décalage géométrique}
On considère les suites $(u_n)_{\nN}$ et $(v_n)_{\nN}$ définie par $u_0=1$ , $u_{n+1}=2(u_n-n)$ et $v_n=2n+2-u_n$
\begin{enumerate}
	\item La suite $(u_n)_{\nN}$ est croissante.
	\item La suite $(u_n)_{\nN}$ est géométrique.
	\item La suite $(v_n)_{\nN}$ est géométrique.
\end{enumerate}



\subsection{Relations entre suites}
On considère une suite $(u_n)_{\nN}$ ainsi que la suite $(v_n)_{\nN}$ définie par $v_n=\frac{2}{3-u_n}$ pour tout entier naturel $n$. On supposera que $u_n\neq3$ pour tout entier naturel $n$
\begin{enumerate}
	\item Si la suite $(u_n)_{\nN}$ est majorée par $2$, alors la suite $(v_n)_{\nN}$ est majorée par $2$
	\item Si la suite $(u_n)_{\nN}$ est majorée, alors la suite $(v_n)_{\nN}$ est majorée.
	\item Si la suite $(v_n)_{\nN}$ est croissante, alors la suite $(u_n)_{\nN}$ est décroissante.
	\item Si la suite $(u_n)_{\nN}$ converge, alors la suite $(v_n)_{\nN}$ converge.
\end{enumerate}



\subsection{Pi carré sur six}
On considère les suites $(u_n)_{n\geq1}$ et $(v_n)_{n\geq1}$ définie par $u_1=1$ , $u_{n+1}=u_n+\frac{1}{(n+1)^2}$ et $v_n=u_n+\frac{1}{n}$
\begin{enumerate}
	\item $v_3=\frac{61}{36}$
	\item La suite $(u_n)_{n\geq1}$ est croissante.
	\item La suite $(v_n)_{n\geq1}$ est croissante.
\end{enumerate}



\subsection{Inverse arithmétique}
Soient $(u_n)_{n\geq2}$ la suite définie par $u_2=2$ et $u_{n+1}=u_n\times(1-\frac{1}{n})$ et $(w_n)_{n\geq2}$ la suite définie par $w_n=\frac{1}{u_n}$\\
On admettra que la suite $(u_n)$ est strictement positive.
\begin{enumerate}
    \item La suite $(u_n)_{n\geq2}$ est croissante
    \item La suite $(u_n)_{n\geq2}$ est majorée
    \item Pour tout entier $n$ supérieur ou égal à $2$, $u_n=\frac{2}{n-1}$
    \item La suite $(w_n)_{n\geq2}$ est arithmétique
\end{enumerate}



\subsection{Somme de suites}
On considère deux suites $(u_n)_{\nN}$ et $(v_n)_{\nN}$ telles que $u_n+v_n=1$ pour tout entier naturel $n$
\begin{enumerate}
	\item Les suites $(u_n)_{\nN}$ et $(v_n)_{\nN}$ convergent.
	\item Si la suite $(u_n)_{\nN}$ est majorée, alors la suite $(v_n)_{\nN}$ est minorée.
	\item Si la suite $(u_n)_{\nN}$ est croissante, alors la suite $(v_n)_{\nN}$ est croissante.
\end{enumerate}



\subsection{Arithmético-géométrique ?}
On considère la suite $(u_n)_{\nN}$ définie par $u_0=1$ et $u_{n+1}=2u_n+n$ pour tout entier naturel $n$
\begin{enumerate}
	\item Pour tout entier naturel $n$, $u_{n+2}=4u_n+3n$
	\item $u_5$ est un multiple de $5$
	\item $u_n=2^{n+1}-2n-1$ pour tout entier naturel $n$
\end{enumerate}



\subsection{Écrasement rapide}
Soit $(u_n)_{\nN}$ la suite définie par $u_0=1$ et $u_{n+1}=\frac{{u_n}^2}{3}$ pour tout entier naturel $n$. On définit la suite $(v_n)_{\nN}$ par $v_n=\ln(u_n)-\ln(3)$ pour tout entier naturel $n$
\begin{enumerate}
	\item La suite $(v_n)_{\nN}$ est géométrique.
	\item Pour tout $n$ entier naturel, $u_n=3^{1-3^n}$
\end{enumerate}



\subsection{Vrac 1}
\begin{enumerate}
	\item Soit $(u_n)_{\nN}$ une suite divergente. S'il existe une suite $(v_n)_{\nN}$ telle que la suite $(u_n\times v_n)_{\nN}$ converge, alors la suite $(v_n)_{\nN}$ converge.
	\item Soit $(u_n)_{\nN}$ la suite définie par $u_n=\dfrac{(3n+1)^3}{3n^3+1}$ pour tout entier naturel $n$. La suite $(u_n)_{\nN}$ converge vers $3$
	\item Soit $(u_n)_{\nN}$ une suite croissante convergeant vers $5$. $\limn \dfrac{5}{5-u_n}=+\infty$
\end{enumerate}



\section*{Quatrième partie - Probabilités}

\subsection{Espérance}
Soit $a$ un nombre réel compris dans l'intervalle $]0~;~1[$ \\
On considère $X$ une variable aléatoire à valeurs dans $\{-1~;~1\}$ vérifiant $\PP(X=1)=a$ \\
On note $\EE(X)$ l'espérance mathématique de la variable $X$
\begin{enumerate}
	\item $\EE(X)=1-2a$
	\item $\EE(x) \geq \frac{1}{2}$ si et seulement si $a\geq\frac{1}{2}$
	\item $\EE(X) \leq a$ pour toute valeur de $a$
\end{enumerate}



\subsection{Lancer de dé}
On paie $2$\euro\, pour lancer un dé équilibré. Si on tombe sur un nombre premier, on gagne le même montant en euros, sinon rien.
\begin{enumerate}
	\item Si le dé possède $6$ faces, ce jeu est rentable.
	\item Si le dé possède $8$ faces, ce jeu est rentable.
	\item Si le dé possède $12$ faces, ce jeu est rentable.
\end{enumerate}



\subsection{Binomiale équilibrée}
Soit $X$ une variable aléatoire suivant une loi binomiale de paramètres $n=20$ et $p=\frac{1}{2}$. On note $\EE(X)$ son espérance mathématique et $\VV(X)$ sa variance.
\begin{enumerate}
	\item $\PP(X=13)=\binom{20}{7}\times\frac{1}{2^{20}}$
	\item $\EE(X)\times\VV(X)=50$
	\item $\PP(X\leq19)=21\times2^{-20}$
\end{enumerate}



\subsection{Évènements}
Soient $A$ et $B$ deux évènement d'un univers $\Omega$
\begin{enumerate}
	\item $\PP\left(\barre{A}\cap B\right)=\PP(B)-\PP(A\cap B)$
	\item $\PP\left(\barre{A}\cap B\right) + \PP\left(A \cap \barre{B}\right) = \PP(A\cup B)$
	\item $\PP\left(\barre{A \cap B}\right) + \PP\left(\barre{A \cup B}\right) = \PP\left(\barre{A}\right)+\PP\left(\barre{B}\right)$
\end{enumerate}



\subsection{Probabilités conditionnelles}
Soient $A$ et $B$ deux évènement d'un univers $\Omega$ tels que $\PP_A(B)=\frac{1}{2}$ , $\PP_B(A)=\frac{1}{3}$ et $\PP(A\cap B)=\frac{1}{6}$
\begin{enumerate}
	\item Les évènements $A$ et $B$ sont indépendants.
	\item $\PP(A\cup B)=\frac{5}{6}$
\end{enumerate}



\subsection{Moments binomiaux}
Soit $X$ une variable aléatoire suivant une loi binomiale de paramètres $n$ et $p$\\
On note $\EE(X)$ son espérance et $\VV(X)$ sa variance.
\begin{enumerate}
    \item Si $p=\frac{1}{2}$ alors $P(X\leq1)=P(X\geq n-1)$
    \item Si $\EE(X)=9$ et $\VV(X)=6$ alors $n=30$
\end{enumerate}



\subsection{Contrôle de l'espérance}
On considère une variable aléatoire $X$ à valeurs dans $\{2~;~4~;~8\}$ et dont l'espérance mathématiques vaut $\EE(X)=5$
\begin{enumerate}
	\item $\PP(X=8)<0,625$
	\item Si $\PP(X=4)=\frac{1}{2}$ alors $\PP(X=8)=\frac{1}{4}$
	\item $\PP(X=2)+\PP(X=4)\leq \PP(X=8)$
\end{enumerate}



\subsection{Vrac 1}
Soient $A$ et $B$ deux évènement d'un univers $\Omega$
\begin{enumerate}
	\item Les évènements $\barre{B}$ , $A \cap B$ et $\barre{A} \cap B$ forment une partition de $\Omega$
	\item Si $\PP(\barre{A})=\frac{2}{3}$ , $\PP_A(B)=\frac{1}{4}$ et $\PP_{\barre{A}}(B)=\frac{1}{2}$ alors $\PP(\barre{B})=\frac{3}{4}$
	\item Si $\PP(A \cup B)=2\PP(A)$ et $\PP(A)=2\PP(A \cap B)$ alors $3\PP(A)=2\PP(B)$
\end{enumerate}



\section*{Cinquième partie - Géométrie du plan}

\subsection{Un triangle particulier}
On considère dans un repère orthonormé les points $A=(3\sqrt{3}~;~3)$ , $B=(\sqrt{3}~;~5)$ et $C=(5\sqrt{3}~;~9)$
\begin{enumerate}
	\item La longueur $AB$ vaut $5$
	\item Le triangle $ABC$ est rectangle en $A$
	\item L'angle $\widehat{ABC}$ a une mesure de $\frac{\pi}{3}$ en radians
\end{enumerate}



\subsection{Vecteurs à paramètre}
Soit $a$ un nombre réel.\\
On considère les vecteurs $\vcoord{u}{\sqrt{2} \\ \sqrt{3}}$ et $\vcoord{v}{a \\ a+1}$
\begin{enumerate}
	\item Si $a=1$, alors $\normev{u}=\normev{v}$
	\item Les vecteurs $\vv{u}$ et $\vv{v}$ sont orthogonaux si et seulement si $a=6-\sqrt{3}$
	\item Les vecteurs $\vv{u}$ et $\vv{v}$ sont colinéaires si et seulement si $a=\sqrt{6}+4$ 
\end{enumerate}



\subsection{Cercle circonscrit}
On considère dans un repère orthonormé les points $A=(4~;~0)$ , $B=(4~;~6)$ , $C=(-1~;~5)$ et $M=(2~;~3)$
\begin{enumerate}
	\item Les points $A$, $B$ et $C$ appartiennent à un même cercle de centre $M$
	\item $\vv{AM}+\vv{BM}+\vv{CM}=\vv{0}$
	\item L'angle $\widehat{BAC}$ mesure $60^\circ$
	\item Le triangle $MBC$ est isocèle rectangle.
\end{enumerate}



\subsection{Autour d'un cercle}
On considère les points $A=(1~;~3)$ et $B=(5~;~1)$ dont les coordonnées sont données dans un repère orthonormé.
\begin{enumerate}
    \item Une équation de la droite $(AB)$ est $x=7-2y$
    \item Une équation du cercle de diamètre $[AB]$ est $x^2+y^2+8=6y+4x$
    \item Il existe exactement deux points du cercle de diamètre $[AB]$ dont l'abscisse est égale à l'ordonnée
    \item Le projeté orthogonal de l'origine du repère sur la droite $(AB)$ est le point de coordonnées $\left(\frac{7}{5}~;~\frac{14}{5}\right)$
\end{enumerate}



\subsection{Produit scalaire}
On considère $A$, $B$ et $C$ trois points du plan rapporté à un repère orthonormé tels que $AB=3$ , $AC=4$ et $\scal{AB}{BC}=-9$
\begin{enumerate}
	\item $BC=5$
	\item L'angle $\widehat{ABC}$ est aigu.
	\item Le triangle $ABC$ est rectangle.
\end{enumerate}



\subsection{Étude d'un triangle}
On munit le plan d'un repère orthonormé $\Oij$.

On considère les points $A=(-2~;~1)$ , $B=(2~;~3)$ et $C=(7~;~-2)$
\begin{enumerate}
	\item Le point $H=(3~;~6)$ est le point de concours des hauteurs du triangle $ABC$.
	\item Le point $G=(2~;~-2)$ est le centre du cercle circonscrit au triangle $ABC$.
	\item L'angle $\widehat{ABC}$ est aigu.
\end{enumerate}



\subsection{Vrac 1}
On munit le plan d'un repère orthonormé $\Oij$.
\begin{enumerate}
	\item Soient $a$, $b$, $c$ et $d$ quatre nombres réels quelconques. Les droites $d_1:y=ax+b$ et\\ $d_2:y=cx+d$ sont perpendiculaires si et seulement si $\frac{a}{c}=1$
	\item Soit un segment $[AB]$ de milieu $I$. Pour tout point $M$ du plan , $\scal{AM}{BM}=MI^2-\frac{1}{4}AB^2$
	\item On considère les points $A=(-3~;~-1)$ , $B=(1~;~5)$ et $C=(8~;~1)$. Le triangle $ABC$ est rectangle.
	\item Une équation du cercle de centre $\Omega=(3~;~-1)$ et de rayon $3$ est $x^2+y^2-6x+2y+1=0$
\end{enumerate}
