\section{Notion de monotonie}

\subsection{Fonctions affines}
Soient $a<b$ deux réels quelconques.
\begin{enumerate}
	\item On considère la fonction $f:x\mapsto 3x-1$ définie sur $\RR$ 
	\begin{enumerate}
		\item Montrer que $f(b)-f(a)=3(b-a)$
		\item Que peut on en déduire concernant la monotonie de la fonction $f$ sur $\RR$ ?
	\end{enumerate}
	\item On considère la fonction $g:x\mapsto 7-2x$ définie sur $\RR$ 
	\begin{enumerate}
		\item Montrer que $g(b)-g(a)=-2(b-a)$
		\item Que peut on en déduire concernant la monotonie de la fonction $g$ sur $\RR$ ?
	\end{enumerate}
\end{enumerate}



\subsection{Première parabole}
On considère la fonction $f$ définie sur $\RR$ par $g(x)=x^2-2x$

Soient $a$ et $b$ deux réels quelconques.
\begin{enumerate}
	\item Montrer que $f(b)-f(a)=(b-a)(a+b-2)$
	\item On suppose que $1<a<b$
	
	\begin{enumerate}
		\item Montrer que $f(b)-f(a)>0$
		\item En déduire que la fonction $f$ est croissante sur l'intervalle $[1;+\infty[$
	\end{enumerate}
	\item On suppose désormais que $a<b<1$
	\begin{enumerate}
		\item Montrer que $f(b)-f(a)<0$
		\item En déduire que la fonction $f$ est décroissante sur l'intervalle $]-\infty;1]$
	\end{enumerate}
\end{enumerate}



\subsection{Une autre ?}
On considère la fonction $\Phi:x\mapsto 2x^2-12x+1$ définie sur $\RR$

Soient $a$ et $b$ deux réels quelconques.
\begin{enumerate}
	\item Montrer que $\Phi(b)-\Phi(a)=2(b-a)(a+b-6)$
	\item On suppose que $3<a<b$
	\begin{enumerate}
		\item Montrer que $\Phi(b)-\Phi(a)>0$
		\item En déduire que la fonction $\Phi$ est croissante sur l'intervalle $[3;+\infty[$
	\end{enumerate}
	\item On suppose désormais que $a<b<3$
	\begin{enumerate}
		\item Montrer que $\Phi(b)-\Phi(a)<0$
		\item En déduire que la fonction $\Phi$ est décroissante sur l'intervalle $]-\infty;3]$
	\end{enumerate}
\end{enumerate}



\subsection{Homographie simple}
On considère la fonction $h:x\mapsto \dfrac{x+1}{x}$
\begin{enumerate}
	\item Déterminer le domaine de définition de la fonction $h$
	\item Pour $a$ et $b$ deux réels non nul, montrer que $h(b)-h(a)=-\dfrac{b-a}{ab}$
	\item En supposant $0<a<b$, monter que la fonction $h$ est décroissante sur l'intervalle $]0;+\infty[$
	\item Quelle est sa monotonie sur l'intervalle $]-\infty;0[$ ?
\end{enumerate}



\subsection{Beaucoup de changements}
On considère la fonction $f:x\mapsto x+\frac{4}{x}$ définie sur $\RR^*$

Soient $a$ et $b$ deux réels non nuls.
\begin{enumerate}
	\item Démontrer que $f(b)-f(a)=\dfrac{(b-a)(ab-4)}{ab}$
	\item Montrer que $f$ est croissante sur l'intervalle $[2;+\infty[$
	\item Montrer que $f$ est décroissante sur l'intervalle $]0;2]$
	\item Montrer que $f$ est croissante sur l'intervalle $[-2;0[$
	\item Montrer que $f$ est décroissante sur l'intervalle $]-\infty;-2[$
\end{enumerate}



\subsection{Une question de parité}
On considère la fonction $\gamma:x\mapsto \dfrac{x^2}{1+x^2}$ définie sur $\RR$
\begin{enumerate}
	\item Démontrer que la fonction $\gamma$ est une fonction paire.
	\item Soient $a$ et $b$ deux réels quelconques.
	\begin{enumerate}
		\item Montrer que $\gamma(b)-\gamma(a)=\dfrac{(b-a)(a+b)}{(1+a^2)(1+b^2)}$
		\item En déduire que la fonction $\gamma$ est croissante sur l'intervalle $[0;+\infty[$
	\end{enumerate}
	\item Sans calculs, justifier alors que $\gamma$ est décroissante sur l'intervalle $]-\infty;0]$
\end{enumerate}



\subsection{Fraction rationnelle}
On considère la fonction $f:x\mapsto\dfrac{x^2+3}{x+1}$ définie sur $\RR\setminus{-1}$

Soient $a$ et $b$ deux réels quelconques distincts de $-1$
\begin{enumerate}
	\item Démontrer que $f(b)-f(a)=\dfrac{(b-a)(ab+a+b-3)}{(a+1)(b+1)}$
	\item On suppose que $1<a<b$
	\begin{enumerate}
		\item Montrer alors que $f(b)-f(a)>0$
		\item En déduire que $f$ est croissante sur un intervalle que l'on précisera.
	\end{enumerate}
	\item On suppose désormais que $-1<a<b<1$
	\begin{enumerate}
		\item Justifier que $f(b)-f(a) >0$
		\item Que peut-on en conclure quant à la monotonie de $f$ sur l'intervalle $]-1;1[$ ?
	\end{enumerate}
\end{enumerate}


\subsection{Degré élevé}
On considère la fonction $\varphi:x\mapsto x^3-3x^2$ définie sur $\RR$

Soient $a$ et $b$ deux nombres réels quelconques.
\begin{enumerate}
	\item Montrer que $\varphi(b)-\varphi(a)=(b-a)(a^2+b^2+ab-3a-3b)$
	\item En déduire que la fonction $\varphi$ est croissante sur l'intervalle $]-\infty;0]$
	\item On suppose que $3<a<b$
	\begin{enumerate}
		\item Justifier que $a^2>3a$ et que $b^2>3b$
		\item En déduire que la fonction $\varphi$ est également croissante sur l'intervalle $[3;+\infty[$
	\end{enumerate}
\end{enumerate}



\section{Notion d'extremum}

\subsection{Première parabole}
On considère la fonction $f:x\mapsto 2x^2-5$ définie sur $\RR$
\begin{enumerate}
	\item Justifier que $f(x)\geq-5$ \quad pour tout $x$ réel.
	\item Peut-on affirmer que $-5$ est le minimum de la fonction $f$ ?
\end{enumerate}



\subsection{Une autre orientation}
On considère la fonction $g:x\mapsto -x^2+4x+1$ définie sur $\RR$
\begin{enumerate}
	\item Démontrer que $g(x)=5-(x-2)^2$ \quad pour tout $x$ réel.
	\item Justifier alors que $g(x)\leq5$ \quad pour tout $x$ réel.
	\item Peut-on affirmer que $5$ est le maximum de la fonction $g$ ?
\end{enumerate}



\subsection{Un bon quotient}
On définit sur $\RR$ la fonction $\varphi$ par $\varphi(x)=\dfrac{2x^2-1}{x^2+1}$
\begin{enumerate}
	 \item \begin{enumerate}
	 	\item Démontrer que $\varphi(x)+1=\frac{x^2}{x^2+1}$
	 	\item En déduire que $\varphi(x)\geq-1$ \quad pour tout $x$ réel.
	 	\item Justifier que $-1$ est en fait le minimum de $\varphi$ sur $\RR$
	\end{enumerate}
	\item \begin{enumerate}
		\item Démontrer que $\varphi(x)-2=\frac{-3}{x^2+1}$ \quad pour tout $x$ réel.
		\item En déduire que $\varphi(x)\leq 2$ \quad pour tout $x$ réel.
		\item Montrer que l'équation $\varphi(x)=2$ n'admet aucune solution réelle.
		\item Peut-on conclure que $2$ est le maximum de $\varphi$ sur $\RR$ ?
	\end{enumerate}
\end{enumerate}



\subsection{Rien qu'un développement}
On considère la fonction $f:x\mapsto\dfrac{2x}{1+x^2}$ définie sur $\RR$
\begin{enumerate}
	\item \begin{enumerate}
		\item En développant l'expression $(x+1)^2$, justifier que $2x\geq-x^2-1$ \quad pour tout $x$ réel.
		\item En développant l'expression $(x-1)^2$, justifier que $2x \leq x^2+1$ \quad pour tout $x$ réel.
	\end{enumerate}
	\item En déduire que $-1\leq f(x) \leq 1$ \quad pour tout $x$ réel.
	\item Peut-on affirmer que $1$  et $-1$ sont des extremums de la fonction $f$ ?
\end{enumerate}
