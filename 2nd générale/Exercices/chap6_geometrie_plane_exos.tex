\section{Géométrie dans les triangles}

\subsection{Avec un parallélogramme}
On considère un parallélogramme $ABCD$. On nomme $E$ le symétrique de $A$ par rapport au point $B$. On note $F$ le point d'intersection entre les droites $(ED)$ et $(BC)$
\begin{enumerate}
	\item Construire une figure afin de représenter la situation.
	\item Démontrer que le point $F$ est le milieu du segment $[BC]$
\end{enumerate}

\subsection{Trapèze droit}
On considère un trapèze $ABCD$ tel que $\widehat{BAD}=\widehat{ADC}=90^\circ$ 
On sait que les diagonales de ce trapèze sont perpendiculaires en un point $E$. 
On sait de plus que $EC=1$ , $DC=2$ et $AE=3$
\begin{enumerate}
	\item Faire un schéma permettant de représenter la situation.
	\item Calculer la longueur $AB$
	\item \begin{enumerate}
		\item Calculer la longueur $EB$
		\item En déduire la longueur $ED$
	\end{enumerate}
	\item Déterminer toutes les longueurs manquantes.
	\item Calculer l'aire du triangle $ABC$
\end{enumerate}



\subsection{En deux coups de sextant}
On considère un triangle $ABC$ rectangle en $B$ tel que $\widehat{ACB}=15^\circ$ 

On place un point $D$  sur le segment $[AC]$ tel que $DC=10$ et $\widehat{ADB}=30^\circ$

\begin{minipage}{0.70\linewidth}
	\begin{enumerate}
		\item Exprimer la longueur $AB$ de deux manière différentes à l'aide des informations données par l'énoncé.
		\item En déduire que $CD\times\tan\left(15^\circ\right)+AD\times\tan\left(15^\circ\right)=AD\times\tan\left(30^\circ\right) $
		\item Montrer que $AD=\dfrac{10\tan\left(15^\circ\right)}{\tan\left(30^\circ\right)-\tan\left(15^\circ\right)}$ puis vérifier à l'aide de la calculatrice que $AD=5\sqrt{3}$
		\item Déterminer alors la longueur $AB$
	\end{enumerate}
\end{minipage}

\subsection{Extension triangulaire}
On considère un triangle $ABC$ de rayon $1$. 

On note $M$ le milieu du segment $[AC]$ et $N$ le symétrique de $B$ par rapport au point $C$ 

On place les points $P$ et $Q$ sur le segment $[AB]$ de sorte que $AP=PQ=QB$
\begin{enumerate}
	\item Faire une figure représentant la situation.
	\item \begin{enumerate}
		\item Justifier que $\frac{AP}{AQ}=\frac{AM}{AC}=\frac{1}{2}$
		\item Que peut-on en déduire concernant les droites $(PM)$ et $(QC)$ ?
	\end{enumerate}
	\item \begin{enumerate}
		\item Justifier que $\frac{BC}{BN}=\frac{BQ}{BP}=\frac{1}{2}$
		\item Que peut-on en déduire concernant les droites $(PN)$ et $(QC)$ ?
	\end{enumerate}
	\item Montrer que les points $M$, $N$ et $P$ sont alignés.
	\item \begin{enumerate}
		\item Quels points appartiennent au cercle $\Ccal$ de centre $C$ et de rayon $1$ ?
		\item En déduire la nature du triangle $ABN$
		\item Calculer la longueur $AN$
	\end{enumerate}
\end{enumerate}
