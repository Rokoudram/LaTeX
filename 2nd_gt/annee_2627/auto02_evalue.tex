\begin{center}
	\underline{\textbf{Correction}}
\end{center}

\begin{enumerate}[itemsep = 3mm]
	\item FAUX : le nombre $(-3)^{-3}$ est négatif , pourtant $-3$ est bien un nombre entier.
	
	\item $\dfrac{6 + \frac{1}{6}}{2} = \dfrac{\frac{6 \times 6}{6} + \frac{1}{6}}{2} = \dfrac{\frac{36}{6} + \frac{1}{6}}{2} = \dfrac{\frac{37}{6}}{2} = \dfrac{37}{12}$
	
	\item $5x + 18 = 13x - 6 \Longleftrightarrow 18 = 8x -6 \Longleftrightarrow 24 = 8x \Longleftrightarrow 3 = x$
	
	\item $\left(\dfrac{2^4}{4^5}\right)^2 = \left(\dfrac{2^4}{(2^2)^5}\right)^2 = \left(\dfrac{2^4}{2^{10}}\right)^2 = (2^{-6})^2 = 2^{-12}$
	
	\item $3^8 \times \dfrac{3^7}{3^{14}} = 3^8 \times 3^{-7} = 3^1 = 3$
	
	\item $\dfrac{2}{3} + \dfrac{3}{4} + \dfrac{4}{5} = 
	\dfrac{2 \times 20}{3 \times 20} + \dfrac{3 \times 15}{4 \times 15} + \dfrac{4 \times 12}{5 \times 12} =
	\dfrac{40}{60} + \dfrac{45}{60} + \dfrac{48}{60} = \dfrac{133}{60} $
\end{enumerate}
