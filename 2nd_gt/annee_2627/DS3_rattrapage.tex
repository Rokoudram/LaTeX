On a représenté dans le repère ci-dessous la courbe $\Ccal_f$ représentative de la fonction $f$ définie sur l'intervalle $[-2 ~;~ 5]$

\begin{minipage}{0.55\linewidth}
    \begin{enumerate} 
    		\item Donner l'ensemble des solutions de l'équation $f(x) = 3$
    		\item Donner l'ensemble des solutions de l'équation $f(x) = -2$
    		\item Donner l'ensemble des solutions de l'inéquation $f(x) > 0$
    		\item Donner l'ensemble des solutions de l'inéquation $f(x) \leq 2$
    		\item Donner l'ensemble des solutions de l'inéquation $f(x) \leq 1$
    \end{enumerate}
\end{minipage}
\begin{minipage}{0.5\linewidth}
    \begin{tikzpicture}[scale=1.1]
        \begin{axis}[
        		xmin = -2 , xmax = 5, ymin = -1, ymax = 3,
			axis lines = middle, 
			grid=both , 
	        	xtick distance = 1, ytick distance = 1,
	    		major grid style = {line width = 0.2pt ,draw = gray},
	    		minor tick num=1 ,
	    		minor grid style = {line width = 0.1pt ,draw = gray!35!white}, 
	        	ticklabel style = {font = \scriptsize } ,
	        	enlargelimits = { abs = 0.2 }
	        	]
    

		\addplot[line width = 2pt, domain=-2:-1, samples=100, color=newblue]
		{3*(1-(x+2))^2*(x+2)*(-1/3) + 3*(1-(x+2))*(x+2)^2*(-1) + (x+2)^3*(-1)};
		
		\addplot[line width = 2pt, domain=-1:0, samples=100, color=newblue]
		{(1-(x+1))^3*(-1) + 3*(1-(x+1))^2*(x+1)*(-1) + 3*(1-(x+1))*(x+1)^2*(-0.5)};
		
		\addplot[line width = 2pt, domain=0:1, samples=100, color=newblue]
		{3*(1-x)^2*x*0.5 + 3*(1-x)*x^2*2 + x^3*2};
		
		\addplot[line width = 2pt, domain=1:2, samples=100, color=newblue]
		{2*(1-(x-1))^3 + 6*(1-(x-1))^2*(x-1) + 3*(1-(x-1))*(x-1)^2 + (x-1)^3};
		
		
		\addplot[line width = 2pt, domain=2:3, samples=100, color=newblue]
		{(1-(x-2))^3 + 3*(1-(x-2))^2*(x-2) + 3*(1-(x-2))*(x-2)^2*1.25 + (x-2)^3*1.5};
		
		
		\addplot[line width = 2pt, domain=3:3.5, samples=100, color=newblue]
		{(1-2*(x-3))^3*1.5 + 3*(1-2*(x-3))^2*2*(x-3)*1.625 + 3*(1-2*(x-3))*(2*(x-3))^2*1.75 + (2*(x-3))^3*2};
		
		\addplot[line width = 2pt, domain=3.5:4, samples=100, color=newblue]
		{(1-2*(x-3.5))^3*2 + 3*(1-2*(x-3.5))^2*2*(x-3.5)*2.25 + 3*(1-2*(x-3.5))*(2*(x-3.5))^2*3 + (2*(x-3.5))^3*3};
		
		\addplot[line width = 2pt, domain=4:5, samples=100, color=newblue]
		{(1-(x-4))^3*3 + 3*(1-(x-4))^2*(x-4)*3 + 3*(1-(x-4))*(x-4)^2};

        
        \draw[color = newblue] (0.5,2.25) node{$\Ccal_f$};
    
        \end{axis}
    \end{tikzpicture}                                               
\end{minipage}
