	\underline{\textbf{\large Correction}}
\end{center}

\begin{enumerate}[itemsep = 3mm]
	\item $\dfrac{6}{5} + \dfrac{2}{3} = \dfrac{6\times3}{5\times3} + \dfrac{2\times5}{3\times5} = \dfrac{18}{15} + \dfrac{10}{15} = \dfrac{28}{15}$
	
	\item $\cfrac{\,\frac{72}{35}\,}{\frac{45}{21}} = \dfrac{72}{35} \times \dfrac{21}{45} = \dfrac{8 \times 9}{7 \times 5} \times \dfrac{7 \times 3}{9 \times 5} = \dfrac{8 \times 3}{5 \times 5} = \dfrac{24}{25}$
	
	\item $\dfrac{2}{3}-\dfrac{x}{6} = \dfrac{1}{2} \Longleftrightarrow 
	\dfrac{2 \times 2}{3 \times 2}-\dfrac{x}{6} = \dfrac{1 \times 3}{2 \times 3} \Longleftrightarrow
	\dfrac{4}{6}-\dfrac{x}{6} = \dfrac{3}{6} \Longleftrightarrow 
	4 - x = 3 \Longleftrightarrow x = 1$
	
	\item $x + 2 = 4x - 7 \Longleftrightarrow 2 = 3x - 7 \Longleftrightarrow 9 = 3x \Longleftrightarrow 3 = x$
	
	\item On commence par calculer la moyenne entre $3$ et $\dfrac{2}{3}$ :
	
	$\dfrac{3+\frac{2}{3}}{2} = \dfrac{\frac{3 \times 3}{3} + \frac{2}{3}}{2} = \dfrac{\frac{9}{3} + \frac{2}{3}}{2} = \dfrac{\frac{11}{3}}{2} = \dfrac{11}{6}$
	
	L'inverse de la moyenne entre $3$ et $\dfrac{2}{3}$ vaut donc $\dfrac{6}{11}$
	
	\item VRAI. Par exemple l'équation $x=x$ admet une infinité de solutions, puisque $x$ est toujours égal à $x$
	\item VRAI : $\dfrac{21 \times 9 \times 12}{27 \times 3 \times 14} = 
	\dfrac{7\times3 \times 9 \times 4\times3}{9\times3 \times 3 \times 2\times7} = 
	\dfrac{4}{2} = 2$
	
	\item $5 - \dfrac{5}{3} = \dfrac{5\times3}{3} -\dfrac{5}{3} = \dfrac{25}{3} - \dfrac{5}{3} = \dfrac{20}{3}$
	
	\item $\cfrac{2+\frac{2}{3}}{2-\frac{2}{3}} = 
	\dfrac{\frac{2\times3}{3} + \frac{2}{3}}{\frac{2\times3}{3} - \frac{2}{3}} =
	\dfrac{\frac{6}{3} + \frac{2}{3}}{\frac{6}{3} - \frac{2}{3}} =
	\dfrac{\frac{8}{3}}{\,\frac{4}{3}\,} = \dfrac{8}{3} \times \dfrac{3}{4} = \dfrac{8}{4} = 2$
	
	\item Notons $x$ le nombre choisi. L'inverse de $x$ est $\dfrac{1}{x}$
	
	On calcule donc $x \times 3 \times \dfrac{1}{x} = \dfrac{x \times 3}{x} = 3$
\end{enumerate}
