	\item $\sqrt{3 + 2\sqrt{2}} = 1 + \sqrt{2}$

\section{Une fonction en morceaux}
On considère la fonction $f$ définie par $f(x) = \abs{2x} - \abs{x+1}$ pour tout nombre $x$ réel.

\subsection{Généralités sur la fonction $f$}
\begin{enumerate}
	\item \begin{enumerate}
		\item Résoudre dans $\RR$ l'équation $2x = x+1$
		\item En déduire un antécédent de $0$ par la fonction $f$.
		\item Vérifier que $-\dfrac{1}{3}$ est un autre antécédent de $0$ par la fonction $f$.
	\end{enumerate}
	\item Peut-on affirmer que la fonction $f$ est positive sur $\RR$ ? Justifier.
\end{enumerate}

\subsection{Calculs nécessaires}
\begin{enumerate}
	\item Résoudre l'équation $-x+1 = 3$ sur l'intervalle $]-\infty ~;~ -1]$
	\item Résoudre l'équation $-3x-1 = 3$ sur l'intervalle $[-1 ~;~ 0]$
	\item Résoudre l'équation $x-1 = 3$ sur l'intervalle $[0 ~;~ +\infty[$
\end{enumerate}

\subsection{Résolution d'une équation}
