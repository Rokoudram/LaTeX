\section{Questions de cours}
Dans cet exercice on note $f$ une fonction quelconque et $\Ccal_f$ sa courbe représentative.
\begin{enumerate}
	\item Si $f(x)=y$, alors quel point appartient à $\Ccal_f$  ?
	\item $\bigstar$ Un nombre peut-il admettre plusieurs antécédents par la fonction $f$ ? Et plusieurs images ?
	\item Qu'est-ce que le domaine de définition de la fonction $f$ ?
\end{enumerate}



\section{Généralités}
On considère la fonction $f$ définie par $f(x) = \dfrac{x^2 + 1}{2x - 4}$
\begin{enumerate}
	\item Calculer l'image de $3$ par la fonction $f$.
	\item Déterminer la valeur interdite de la fonction $f$.
	\item $\bigstar$ Donner alors $\Dcal_f$ le domaine de définition de la fonction $f$.
\end{enumerate}

\section{Questions de cours}
Dans cet exercice on note $f$ une fonction quelconque et $\Ccal_f$ sa courbe représentative.
\begin{enumerate}
	\item Si $f(x)=y$, alors le point de coordonnées $(x;y)$ appartient à $\Ccal_f$
	\item $\bigstar$ Un nombre peut admettre plusieurs antécédents par la fonction $f$, mais pas plusieurs images.
	\item Le domaine de définition de la fonction $f$ est l'ensemble des nombres $x$ pour lesquels on peut calculer $f(x)$
\end{enumerate}



\section{Généralités}
On considère la fonction $f$ définie par $f(x) = \dfrac{x^2 + 1}{2x - 4}$
\begin{enumerate}
	\item $f(3) = \dfrac{3^2 + 1}{2 \times 3 - 4} = \dfrac{10}{2} = 5$ \quad donc l'image de $3$ par la fonction $f$ est $5$.
	\item Il est interdit de diviser par $0$, donc $2x - 4 \neq 0$ c'est à dire $x \neq 2$ \\
	La valeur interdite de la fonction $f$ est donc $2$.
	\item $\bigstar$ Le domaine de définition de la fonction $f$ est l'ensemble de tous les nombres réels sauf $2$. \\
	Donc $\Dcal_f = ]-\infty ~;~ 2[ \bbigcup ]2 ~;~ +\infty[$
\end{enumerate}



\section{Représentation graphique}
On considère la fonction $g$ définie par $g(x) = \dfrac{(2-x)(x+1)(x+4)}{4}$ et on note $\Ccal_g$ sa courbe représentative.

\begin{minipage}{0.5\textwidth} \restoreskip
	\begin{enumerate}
		\item On n'essaie jamais de diviser par $0$, et il n'y a pas de racine carrée, donc $\Dcal_g = \RR$
		\item On calcule $g(3) = \dfrac{(2-3)(3+1)(3+4)}{4} = -7$ \quad donc il existe bien un point d'abscisse $3$ appartenant à $\Ccal_g$ et ses coordonnées sont $(3 ~;~ -7)$
		\item $\Ccal_g$ passe par l'origine du repère seulement si $g(0) = 0$
		
		Or $g(0) = \dfrac{(2 - 0)(0 + 1)(x + 0)}{4} = 2$
		
		Donc $\Ccal_g$ ne passe pas par l'origine du repère.
		\item		
		\renewcommand{\arraystretch}{1.5}
		\begin{tabular}{| P{1.5cm} | P{1cm} | P{1cm} | P{1cm} | P{1cm} |} \hline
			$x$ & $-4$ & $-3$ & $-2$ & $-1$  \\ \hline
			$g(x)$ & $0$ & $-2,5$ & $-2$ & $0$ \\ \hline
		\end{tabular}\\
		\begin{tabular}{| P{1.5cm} | P{1cm} | P{1cm} | P{1cm} |} \hline
			$x$ & $0$ & $1$ & $2$  \\ \hline
			$g(x)$ & $2$ & $2,5$ & $0$ \\ \hline
		\end{tabular} \\
		
		On détaille : 
		\begin{align*}
			g(-4) = & \dfrac{(2 - (-4) )(-4 + 1)(-4 + 4)}{4} \\
			& = \dfrac{6 \times (-3) \times 0}{4} = 0
		\end{align*}
	\end{enumerate}
\end{minipage}
