\subsection{Quelques opérations}
On munit le plan du repère $\Oij$ et on considère les vecteurs $\vcoord{u}{-3\\4}$ et $\vcoord{v}{2\\5}$\\
Soit $\alpha$ un nombre réel.
\begin{enumerate}
	\item Déterminer les coordonnées des vecteurs $\vv{x}=2\vv{u}-\vv{v}$ \quad ; \quad $\vv{y}=3\vv{u}-4\vv{v}$ \quad et \quad $\vv{z}=2\vv{u}+3\vv{v}$
	\item Pour quelle(s) valeur(s) de $\alpha$ le vecteurs $\vv{u}+\alpha\vv{v}$ a-t-il une ordonnées nulle ?
	\item Pour quelle(s) valeur(s) de $\alpha$ le vecteur $\alpha\vv{u}+\vv{v}$ a-t-il une abscisse et une ordonnée égales ? 
\end{enumerate}



\subsection{Sur une droite}
On munit le plan du repère $\Oij$ et on considère les points $A\smeq(2~;~1)$ et $B\smeq(-3~;~3)$
\begin{enumerate}
	\item Déterminer les coordonnées du vecteur $\vv{AB}$
	\item Déterminer par le calcul les coordonnées du point $M$ vérifiant l'égalité $\vv{AM}=\vv{BA}$
	\item Déterminer par le calcul les coordonnées du point $N$ vérifiant l'égalité $\vv{BN}=2\vv{BA}$
	\item Vérifier que $\vv{AM}+\vv{BN}=\vv{AB}$
\end{enumerate}



\subsection{Équations vectorielles}
On munit le plan du repère $\Oij$ et on considère les points $A\smeq(-1~;~-2)$ , $B\smeq(5~;~1)$ et $C\smeq(2~;~4)$
\begin{enumerate}
	\item Déterminer les coordonnées des vecteurs $\vv{AB}$ , $\vv{AC}$ et $\vv{BC}$
	\item Déterminer les coordonnées du point $I$ vérifiant $\vv{AI}=\vv{AB}+\vv{AC}$
	\item Déterminer les coordonnées du point $J$ vérifiant $\vv{BJ}=\frac{2}{3}\vv{CA}$
	\item Déterminer les coordonnées du point $K$ vérifiant $\vv{CK}=\frac{1}{3}\vv{BA}+\vv{BJ}$
	\item Déterminer les coordonnées du point $L$ vérifiant $\vv{KL}=\frac{2}{3}\vv{BI}-\vv{AJ}$
\end{enumerate}



\subsection{Encore une fois}
On munit le plan du repère $\Oij$ et on considère les points $A\smeq(-5~;~4)$ , $B\smeq(3~;~2)$ et $C\smeq(-1~;~-3)$
\begin{enumerate}
	\item Déterminer les coordonnées du point $M$ vérifiant $\vv{AM}+\vv{BM}=\vv{CM}$
	\item Déterminer les coordonnées du point $N$ vérifiant $2\vv{MN}=\vv{AB}$
	\item Déterminer les coordonnées du point $G$ vérifiant $\vv{AG}+\vv{BG}+\vv{CG}=\vv{0}$
	\item Déterminer les coordonnées du point $P$ vérifiant $\frac{3}{4}\vv{PG}+\frac{1}{2}\vv{PA}=\vv{MG}$
\end{enumerate}



\subsection{Vecteur paramétré}
Soit $\alpha$ un nombre réel quelconque.\\
On munit le plan du repère $\Oij$ et on considère les vecteurs $\vcoord{u}{\alpha\\1}$ , $\vcoord{v}{1\\-\alpha}$ et $\vcoord{w}{-\alpha\\ \alpha}$

Existe-t-il une valeur de $\alpha$ pour laquelle le vecteur $\alpha\vv{u}+\vv{v}+\vv{w}$ a une abscisse et une ordonnée égales ?

\subsection{Quadrilatère}
On munit le plan du repère $\Oij$ et on considère les points $A\smeq(2~;~5)$ , $B\smeq(-1~;~1)$ , $C\smeq(1~;~2)$ et $D\smeq(6~;~-1)$
\begin{enumerate}
	\item Les vecteurs $\vv{AB}$ et $\vv{CD}$ sont-ils colinéaires ?
	\item Les vecteurs $\vv{AD}$ et $\vv{BC}$ sont-ils colinéaires ?
	\item Les vecteurs $\vv{AC}$ et $\vv{BD}$ sont-ils colinéaires ?
	\item Quelle est la nature du quadrilatère $ABCD$ ? 
\end{enumerate}



\subsection{Alignement}
On munit le plan du repère $\Oij$ et on considère les points $A\smeq(-4~;~-2)$ , $B\smeq(-1~;~-1)$ et $C\smeq(5~;~1)$

Les points $A$, $B$ et $C$ sont-ils alignés ?



\subsection{Sur-alignement}
On munit le plan du repère $\Oij$ et on considère les points $M\smeq(-4~;~8)$ , $N\smeq(10~;~-13)$ , $P\smeq(4~;~-4)$ et $Q\smeq(-6~;~11)$

Démontrer que les points $M$ , $N$ , $P$ et $Q$  sont alignés.



\subsection{Construction}
On munit le plan du repère $\Oij$ et on considère les points $A\smeq(-3~;~2)$ , $B\smeq(1~;~4)$ et $C\smeq(2~;~1)$
\begin{enumerate}
	 \item Déterminer les coordonnées du point $I$ vérifiant $\vv{CI}=2\vv{CA}+3\vv{BC}$
	 \item Déterminer les coordonnées du point $J$ vérifiant $2\vv{AJ}=\vv{AB}+2\vv{CB}$
	 \item Les points $A$, $I$ et $J$ sont-ils alignés ?
\end{enumerate}



\subsection{Le long d'une ligne}
On munit le plan du repère $\Oij$ et on considère les points $A\smeq(2~;~2)$ , $B\smeq(4~;~5)$ et $C\smeq(8~;~1)$
\begin{enumerate}
	 \item Déterminer les coordonnées du point $M$ vérifiant $\vv{CM}=\frac{1}{2}\vv{BM}+\vv{CA}$
	 \item Déterminer les coordonnées du point $N$ vérifiant $\vv{BN}=\vv{MN}-\vv{AN}$
	 \item Montrer que les points $A$ , $B$ , $M$ et $N$ sont alignés.
\end{enumerate}



\subsection{Colinéarité paramétrée}
On munit le plan du repère $\Oij$ et on considère $\alpha$ un nombre réel quelconque.
\begin{enumerate}
	\item Pour quelle(s) valeur(s) de $\alpha$ les vecteurs $\vcoord{u}{\alpha\\3}$ et $\vcoord{v}{-3\\5}$ sont-ils colinéaires ?
	\item Justifier qu'il n'existe aucune valeur de $\alpha$ pour laquelle les vecteurs $\vcoord{x}{\alpha\\2}$ et $\vcoord{y}{-1\\2\alpha}$ sont colinéaires.
	\item Pour quelle(s) valeur(s) de $\alpha$ les vecteurs $\vcoord{u}{\alpha\\6}$ et $\vcoord{v}{3\\2\alpha}$ sont-ils colinéaires ?
\end{enumerate}



\subsection{Alignement incertain}
On munit le plan du repère $\Oij$ et on considère $\alpha$ un nombre réel quelconque.

On considère les points $A\smeq(\alpha~;~1)$ , $B\smeq(2\alpha+1~;2-\alpha)$ et $C\smeq(6~;~\alpha+1)$
\begin{enumerate}
	\item Déterminer les coordonnées des vecteurs $\vv{AB}$ et $\vv{AC}$
	\item Existe-t-il une valeur de $\alpha$ pour laquelle les points $A$, $B$ et $C$ sont alignés ?
\end{enumerate}
