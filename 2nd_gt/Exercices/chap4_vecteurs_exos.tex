On considère un parallélogramme $ABCD$. \\
On note $I$ et $J$ les milieux respectifs des segments $[AD]$ et $[BC]$
\begin{enumerate}
	\item Donner tous les vecteurs égaux à $\vv{IA}$
	\item Quelle est la nature du quadrilatère $AIJB$ ? Et de $IDCJ$ ?
	\item \begin{enumerate}
		\item Quelle est la nature du quadrilatère $IBJD$ ?
		\item En déduire que $\vv{BI}=\vv{JD}$
	\end{enumerate}
	\item \begin{enumerate}
		\item Quelle est la nature du quadrilatère $ICJA$ ?
		\item En déduire que $\vv{IC}=\vv{AJ}$
	\end{enumerate}
	\item On note $M$ le point d'intersection entre les droites $(IC)$ et $(JD)$ \\
	On note $P$ le point d'intersection entre les droites $(IB)$ et $(JA)$
	\begin{enumerate}
		\item Justifier que $P$ et $M$ sont les milieux respectifs des segments $[IB]$ et $[DJ]$
		\item En déduire alors que $\vv{IP}=\vv{MJ}$
		\item Que peut-on conclure quant au quadrilatère $IPJM$ ?
	\end{enumerate}

\subsection{À partir d'un segment}

\subsection{Simplification}

\subsection{Milieu d'un segment}
\begin{enumerate}
	\item On considère un segment $[AB]$ de milieu $I$, ainsi qu'un point $M$ quelconque. \\
	Démontrer que $\vv{MA}+\vv{MB}=2\vv{MI}$
	\item On considère un parallélogramme $CDEF$ ainsi qu'un point $M$ quelconque. \\
	En appliquant le résultat de la question 1. démontrer que $\vv{MC}+\vv{ME}=\vv{MD}+\vv{MF}$
	\item On considère un triangle $GHK$\\
	On note $G'$ le milieu de $[HK]$, $H'$ le milieu de $[GK]$ et $K'$ le milieu de $[GH]$ \\
	En appliquant le résultat de la question 1. démontrer que $\vv{GG'}+\vv{HH'}+\vv{KK'}=\vv{0}$
\end{enumerate}



\subsection{Décomposition}
On considère quatre points du plan $A$, $B$, $C$ et $D$. On pose également $\vv{u}=3\vv{AB}+2\vv{CD}+4\vv{CB}$
\begin{enumerate}
	\item Montrer que $\vv{u}=2\vv{AD}+6\vv{CB}+\vv{AB}$
	\item Montrer que $\vv{u}=6\vv{CD}+7\vv{DB}+3\vv{AD}$
	\item Montrer que $\vv{u}=\vv{AB}+2\vv{AD}$
\end{enumerate}

	\item Montrer que $\vv{AC}+3\vv{JB}+\vv{BD}=\vv{JC}$
	\item Montrer que $2\vv{IJ}-\vv{AB}+\vv{DI}=\vv{IB}$
\end{enumerate}


\subsection{Découpage du triangle}
On considère un triangle $ABC$. \\
On note $I$ le milieu de $[AC]$, $J$ le milieu de $[BC]$ et $K$ le milieu de $[AB]$ 
\begin{enumerate}
	\item Montrer que $\vv{IC}+\vv{JB}+\vv{KA}=\vv{0}$
	\item \begin{enumerate}
		\item Montrer que $\vv{IK}=\frac{1}{2}\vv{CA}+\frac{1}{2}\vv{AB}$
		\item En déduire que $\vv{IK}=\vv{JB}$
		\item Quelle est la nature du quadrilatère $IKBJ$ ?
	\end{enumerate}
	\item On note $M$ le milieu du segment $[IJ]$
	\begin{enumerate}
		\item Montrer que $\vv{IM}=\frac{1}{4}\vv{AB}$
		\item En déduire que $\vv{CM}=\frac{1}{2}\vv{AC}+\frac{1}{4}\vv{AB}$
		\item Démontrer alors que $\vv{MK}=\vv{CM}$
		\item Que peut on en déduire concernant la nature du quadrilatère $CIKJ$ ?
	\end{enumerate}
\end{enumerate}



\subsection{Extension d'un triangle}
On considère un triangle $ABC$ et on définit les points $M$, $N$ et $P$ vérifiant les relations :\\
$\vv{BM}=2\vv{AB}+\vv{CB}$ \quad ,\quad $\vv{CN}=2\vv{AB}+2\vv{AC}$ \quad et \quad $\vv{CP}=3\vv{AB}$
\begin{enumerate}
	\item Démontrer que $\vv{NP}=\vv{AB}-2\vv{AC}$
	\item \begin{enumerate}
		\item Montrer que $\vv{MP}=\vv{AB}+2\vv{BC}$
		\item En déduire que $\vv{MP}=2\vv{AC}-\vv{AB}$
	\end{enumerate}
	\item Que peut-on dire du point $P$ par rapport au segment $[MN]$
\end{enumerate}



\section{Colinéarité et alignement}

\subsection{Relation de colinéarité}
On considère quatre points distincts $A$, $B$, $C$ et $D$.\\
Dans chacune des situations suivantes, montrer que les vecteurs $\vv{u}$ et $\vv{v}$ sont colinéaires :

a) $\vv{u}=2\vv{AB}-\vv{AC}$ \qquad et \qquad $\vv{v}=6\vv{AB}-3\vv{AC}$

b) $\vv{u}=3\vv{BC}+\vv{AD}$ \qquad et \qquad $\vv{v}=6\vv{BC}-2\vv{DA}$

c) $\vv{u}=4\vv{AB}-2\vv{BD}$ \qquad et \qquad $\vv{v}=3\vv{BD}+6\vv{BA}$

d) $\vv{u}=2\vv{AB}+\vv{BC}$ \qquad et \qquad $\vv{v}=2\vv{AB}+2\vv{AC}$

e) $\vv{u}=4\vv{AC}+2\vv{AD}$ \qquad et \qquad $\vv{v}=3\vv{AC}+3\vv{CD}$

f) $\vv{u}=3\vv{BD}+2\vv{DC}$ \qquad et \qquad $\vv{v}=9\vv{BC}-3\vv{DC}$ 



\subsection{Alignement}
On considère un parallélogramme $EFGH$ et on définit les points $I$ et $J$ par  $\vv{EI}=2\vv{EH}$ et $\vv{FJ}=2\vv{EF}-\vv{EH}$
\begin{enumerate}
	\item Montrer que $\vv{GI}=\vv{FH}$
	\item Montrer que $\vv{GJ}=2\vv{HF}$
	\item que peut-on en conclure concernant les points $G$, $I$ et $J$ ?
\end{enumerate}



\subsection{Prolongement triangulaire}
On considère un triangle $ABC$, ainsi que les points $K$ et $L$ vérifiant $\vv{CK}=\vv{AB}+\vv{AC}$ et $\vv{AL}=\vv{CB}$

Montrer que les points $K$, $B$ et $L$ sont alignés.



\subsection{Brochette}
On considère triangle $ABC$ et on définit les points $I$, $J$, $K$ et $L$ par :

$\vv{AJ}=\vv{CB}$ \quad , \quad $\vv{BI}=\vv{AC}$ \quad , \quad $\vv{BK}=\vv{CB}$ \quad et \quad $\vv{CL}=\vv{AI}$
\begin{enumerate}
	\item Justifier que les points $C$, $B$ et $K$ sont alignés.
	\item \begin{enumerate}
		\item Montrer que $\vv{CI}=\vv{AB}$ et $\vv{JK}=\vv{AB}$
		\item Quelle est la nature du quadrilatère $CIKJ$ ?
	\end{enumerate}
	\item Démontrer que les points $I$ , $J$, $B$ et $L$ sont alignés.
\end{enumerate}



\subsection{Parallélisme}
On considère un triangle $IJK$ ainsi que les points $M$ et $N$ tels que $\vv{IM}=\vv{KI}$ et $\vv{KN}=\vv{JK}+2\vv{JI}$

Démontrer que les droites $(NM)$ et $JK$ sont parallèles.



\subsection{Base triangulaire}
On considère triangle $ABC$ et on définit les points $M$, $N$, et $P$ par $\vv{AM}=3\vv{BC}$, $\vv{CN}=\vv{BC}$ et $\vv{BP}=2\vv{AB}$
\begin{enumerate}
	\item Montrer que $\vv{MP}=3\vv{CB}+3\vv{AB}$
	\item \begin{enumerate}
		\item Montrer que $\vv{MN}=2\vv{CB}+3\vv{AC}$
		\item En déduire que $\vv{MN}=\vv{AB}+\vv{CB}$
	\end{enumerate}
	\item Que dire des points $M$, $N$ et $P$ ?
\end{enumerate}
