\subsection{Parité par étude algébrique}
\begin{enumerate}
	\item Démontrer que les fonctions suivantes sont paires : \\	
	\begin{tabular}{p{0.3\linewidth} p{0.3\linewidth} p{0.3\linewidth}}
		$f_1:x\mapsto 5-x^2$ & $f_2:x\mapsto \dfrac{1+x^2}{2+x^2}$ & $f_3:x\mapsto \dfrac{(3-x)(3+x)}{7+2x^4}$	
	\end{tabular}
	\item Démontrer que les fonctions suivantes sont impaires : \\	
	\begin{tabular}{p{0.3\linewidth} p{0.3\linewidth} p{0.3\linewidth}}
		$f_1:x\mapsto 3x(x^2+1)$ & $f_2:x\mapsto \dfrac{(1+x)(1-x)}{x^3}$ & $f_3:x\mapsto \dfrac{5}{2x}-\dfrac{x}{3+x^4}$	
	\end{tabular}
	\item Démontrer que les fonctions suivantes ne sont ni paires ni impaires: \\	
	\begin{tabular}{p{0.3\linewidth} p{0.3\linewidth} p{0.3\linewidth}}
		$f_1:x\mapsto 1+x-x^2$ & $f_2:x\mapsto \dfrac{3+x}{2+x^2}$ & $f_3:x\mapsto x^3+2x^2-x+1$	
	\end{tabular}
\end{enumerate}



\subsection{Une fonction presque impaire}
On considère la fonction $f:x\mapsto \dfrac{1-2x}{x}$ définie sur $\RR^*$
\begin{enumerate}
	\item Montrer que la fonction $x$ n'est ni paire ni impaire.
	\item Pour $a$ un certain réel, on définit la fonction $g$ par $g(x)=f(x)+a$
	\begin{enumerate}
		\item Montrer que $g(-x)=\dfrac{ax-1-2x}{x}$
		\item Justifier qu'il existe une valeur de $a$ pour laquelle la fonction $g$ est paire puis la donner.
	\end{enumerate}
\end{enumerate}
