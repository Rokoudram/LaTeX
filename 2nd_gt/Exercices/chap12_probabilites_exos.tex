\begin{itemize}[label=$\bullet$] 
	\item 20 cartes bleues, numérotées entre 0 et 9 (2 de chaque numéro)
	\item 20 cartes rouges, numérotées entre 0 et 9 (2 de chaque numéro)
	\item 20 cartes jaunes, numérotées entre 0 et 9 (2 de chaque numéro)
	\item 20 cartes vertes, numérotées entre 0 et 9 (2 de chaque numéro)
	\item 8 cartes \og +2 \fg \, (2 de chaque couleur)
	\item 8 cartes \og INVERSION \fg \, (2 de chaque couleur)
	\item 8 cartes \og PASSER UN TOUR \fg \, (2 de chaque couleur)
	\item 4 cartes \og JOKER \fg

	\item Combien de cartes un jeu complet de UNO comporte-t-il ?
	\item Quelle est la probabilité que la carte tirée soit une carte spéciale ?
	\item Quelle est la probabilité que la carte tirée soit de couleur verte ?
	\item Quelle est la probabilité que la carte tirée porte un numéro ?
	\item Quelle est la probabilité que la carte tirée puisse être jouée par-dessus un 7 de couleur jaune ?

On considère $A$ et $B$ deux évènements d'un univers $\Omega$
\begin{enumerate} \setlength{\itemsep}{1mm}
	\item Si $\PP(A)=\frac{1}{3}$ , $\PP(B)=\frac{1}{2}$ et $\PP(A \cap B)=\frac{1}{6}$ , quelle est la valeur de $\PP(A \cup B)$ ?
	\item Si $\PP(A)=\frac{2}{5}$ , $\PP(B)=\frac{1}{15}$ et $\PP(A \cup B)=\frac{2}{3}$ , quelle est la valeur de $\PP(A \cap B)$ ?
	\item Si $\PP(A)=\frac{1}{2}$ , $\PP(A \cup B)=\frac{3}{4}$ et $\PP(A \cap B)=\frac{1}{8}$ , quelle est la valeur de $\PP(B)$ ?
\end{enumerate}
