\begin{enumerate}[label=\underline{Question \no\arabic*} :  , leftmargin=* , itemsep=5mm]
	\item $u_4 = 4^2 - 2\times4 + 2 = 16 - 8 + 2 = 10$
	
	\item On applique la formule du crible : $\PP(A) + \PP(B) = \PP(A \cap B) + \PP(A \cup B)$ \\
	Donc $0,3 + \PP(B) = 0,4 + 0,5$ \quad donc \quad $\PP(B) = 0,9 - 0,3 =0,6$ 
	
	\item La suite $(v_n)$ est arithmétique de raison $-2$, donc chaque terme vaut $2$ de moins que le précédent.
	
	On obtient donc : $v_1 = 5 \quad v_2 = 3 \quad v_3 = 1$ \quad et \quad $v_4 = -1$
	
	\item 
	$\begin{WithArrows}[format = cc]
    & \dfrac{x}{2} + 15 = x - 27 \Arrow{On ajoute $27$ de chaque côté} \\
    \Leftrightarrow  & \dfrac{x}{2} + 42 = x \Arrow{On soustrait $\dfrac{x}{2}$ de chaque côté} \\
    \Leftrightarrow  & 42 = \dfrac{x}{2} \Arrow{On multiplie par $2$ de chaque côté} \\
    \Leftrightarrow  & 84 = x
    \end{WithArrows}$
    
    \item On note $P$ le prix de départ. Une diminution de $30\%$ revient à une multiplication par $0,7$.
    
    On sait donc que $P \times 0,7 = 49$
    
    Donc $P = \dfrac{49}{0,7} = \dfrac{490}{7} = \dfrac{7 \times 70}{7} = \dfrac{\not 7 \times 70}{\not 7} = 70$
    
    \item $f(3) = \dfrac{4}{3 + 1} - \sqrt{3 + 1} = \dfrac{4}{4} - \sqrt{4} = 1 - 2 = -1 $
\end{enumerate}
