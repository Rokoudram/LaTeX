\begin{enumerate}[label=\underline{Question \no\arabic*} :  , leftmargin=* , itemsep=5mm]
	\item On calcule : $\dfrac{350}{100} \times 40 = \dfrac{35 \times 4 \times 100}{100}
	= \dfrac{35 \times 4 \times \not{100}}{\not{100}} = 35 \times 4 = 140$
	
	\item On applique la formule : $\PP_A(B) = \dfrac{\PP(A \cap B)}{\PP(A)} = \dfrac{\,\frac{1}{3}}{\frac{2}{3}} = \dfrac{1}{3} \times \dfrac{3}{2} = \dfrac{1}{\not3} \times \dfrac{\not3}{2} = \dfrac{1}{2} = 0,5$
	
	\item Notons $P$ le prix initial. Le prix $P$ a diminué de $40\%$, donc il a été multiplié par $0,6$ \\
	On peut donc écrire : $P \times 0,6 = 48$ , ainsi $P = \dfrac{48}{0,6} = \dfrac{480}{6} = 80$
	
	\item 
	$\begin{WithArrows}[format = cc]
    & 7x + 31 = 79 - 5x \Arrow{On ajoute $5x$ de chaque côté} \\
    \Leftrightarrow  & 12x + 31 = 79 \Arrow{On soustrait $31$ de chaque côté} \\
    \Leftrightarrow  & 12x = 48 \Arrow{On divise par $12$ de chaque côté} \\
    \Leftrightarrow  & x = \dfrac{48}{12}  = 4
    \end{WithArrows}$
    
    \item En appliquant la formule du crible on obtient : $\PP(B) + 0,7 = 0,3 + 0,9$ , donc $\PP(B) = 0,1$
    
    \item $f(4) = 4^2 - \sqrt{4} - 4 = 16 - 2 - 4 = 10$
\end{enumerate}
