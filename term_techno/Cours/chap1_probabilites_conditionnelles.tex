\section{Rappels}

\begin{rappel}{Probabilités conditionnelles}
	On considère deux évènements $A$ et $B$ tels que $\PP(A) \neq 0$
	
	Alors on note $\PP_A(B)$ la \textbf{probabilité} de l'évènement $B$ \textbf{sachant que} l'évènement $A$ est réalisé.
	
	On calcule cette probabilité grâce à la formule $\PP_A(B) = \dfrac{\PP(A \cap B)}{\PP(A)}$
\end{rappel}

\begin{remarque}
	Cette formule fonctionne bien sûr "à l'envers" et on peut calculer $\PP_B(A) = \dfrac{\PP(A \cap B)}{\PP(B)}$
\end{remarque}



Des deux formules précédentes, on voit que peu importe quel évènement impose la condition, la probabilité de l'évènement $A \cap B$ intervient dans les deux cas. Cela veut donc dire que l'on peut calculer $\PP(A \cap B)$ à partir des probabilités conditionnelles !

\begin{propriete}{Intersection}
	On considère deux évènements $A$ et $B$ de probabilité non nulle. On a alors :
	\[ \PP(A \cap B) = \PP_A(B) \times \PP(B) = \PP_B(A) \times \PP(A) \]
\end{propriete}



\section{Arbres pondérés de probabilités}

Un moyen clair de représenter l'influence des évènements $A$ et $B$ l'un sur l'autre est d'utiliser un arbre pondéré de probabilités. C'est un arbre plus complet que les arbres des possibilités que vous avez manipulé en classe de seconde.

\begin{definition}{Arbre pondéré}
	\begin{minipage}{0.5\linewidth} \restoreskip \vspace{0pt}
		Si l'on commence par étudier l'évènement $A$, alors on places les évènements $A$ et son contraire $\barre{A}$ au bout des premières branches, ainsi que les probabilités associées sur les branches.
		
		À la suite on représente les évènements $B$ et $\barre{B}$ sachant que l'évènement $A$ est réalisé ou non.
		
		En bout de branche, on obtient alors l'intersection entre ces évènements.
	\end{minipage} \hspace{5mm}
	\begin{minipage}{0.5\linewidth}
		\begin{tikzpicture}[
			grow=right, scale = 0.8 ,
			level 1/.style={level distance=4cm, sibling distance=3cm},
			level 2/.style={level distance=3cm, sibling distance=1.5cm},
			level 3/.style={level distance=2cm, sibling distance=3cm}
			]
			\small
			
			\node {$\Omega$}
			child {node {$\barre{A}$} 
				child {node {$\barre{B}$}
					child {node {$\barre{A}\cap\barre{B}$}
					} edge from parent node[below] {$P_{\barre{A}}(\barre{B})$}
				} 
				child {node {$B$}
					child {node {$\barre{A}\cap B$}
					} edge from parent node[above] {$P_{\barre{A}}(B)$}
				} edge from parent node[below] {$P(\barre{A})$}
			}
			child {node {$A$}
				child {node {$\barre{B}$}
					child {node {$A \cap \barre{B}$}
					} edge from parent node[below] {$P_A(\barre{B})$}
				}
	    		child{node {$B$}
	    			child {node {$A \cap B$}
					} edge from parent node[above] {$P_A(B)$}
	    		} 	edge from parent node[above] {$P(A)$}
	    	};
		\end{tikzpicture}
	\end{minipage}
\end{definition}

\begin{astuce}
	En reprenant la formule $\PP(A \cap B) = \PP_A(B) \times \PP(B) = \PP_B(A) \times \PP(A)$, on voit que pour obtenir une probabilité situé au bout de l'arbre, il faut multiplier les probabilités situées sur les branches précédentes !
\end{astuce}



\section{Indépendance d'évènements}

\begin{definition}{Évènements indépendants}
	On dit que deux évènements $A$ et $B$ sont \textbf{indépendants} si la réalisation de l'un n'a pas d'influence sur la réalisation de l'autre.
	
	Autrement dit si $P_A(B)=P(B)$ \quad et \quad $P_B(A)=P(A)$
\end{definition}



\begin{propriete}{Indépendance et intersection}
	Deux évènements $A$ et $B$ sont indépendants si et seulement si $P(A \cap B)=P(A) \times P(B)$
\end{propriete}
