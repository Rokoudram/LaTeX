\section{Partitions de l'univers}

\begin{definition}{Partition de l'univers}
	Soit un univers $\Omega$
	
	On dit que les évènements $A_1~;~A_2~;~\ldots~;~A_n$ forment une \textbf{partition de l'univers} $\Omega$ si :
	\begin{itemize}[label=$\bullet$]
		\item Ces évènements sont deux à deux incompatibles.
		\item Leur réunion est égale à $\Omega$
	\end{itemize}
\end{definition}

\section{Loi des probabilités totales}

\begin{theoreme}{Loi des probabilités totales}
	Soient $(A_1~;~A_2~;~\ldots~;~A_n)$ formant une partition de l'univers. Soit $B$ un évènement quelconque.
	
	On peut calculer la probabilité de l'évènement $B$ par la formule : \\
	\phantom{}\hspace{25mm} $ \PP(B) = \PP(A_1 \cap B) + \PP(A_2 \cap B) + \ldots + \PP(A_n \cap B) \left( = \sommek{0}{n} \PP(A_k \cap B)\right) $
\end{theoreme}



\begin{remarque}
	Les évènements $A$ et $\barre{A}$ formant toujours une partition de l'univers, on a : $\PP(B)=\PP(A \cap B) + \PP(\barre{A} \cap B)$
\end{remarque}
