\definition{Taux d'évolution}{
Le \textbf{taux d'évolution} est un nombre réel permettant de décrire un changement dans la valeur d'une grandeur ou d'une quantité.

Une hausse de $x\%$ correspond à un taux d'évolution $t=\frac{x}{100}$\\
Une diminution de $x\%$ correspond à un taux d'évolution $t=\frac{-x}{100}$
}

\definition{Coefficient multiplicateur}{
On définit le \textbf{coefficient multiplicateur} correspondant à l'évolution d'une valeur initiale ($V_I$) en une valeur finale ($V_F$) par :
\[ CM = \frac{V_F}{V_I} \]
}

\theoreme{$CM$ et taux}{
Le coefficient multiplicateur et le taux d'évolution sont liés par une formule simple :
$CM=1+t$
}


\theoreme{Évolutions successives}{%
Lorsqu'une grandeur subit plusieurs \textbf{évolutions successives} de coefficients multiplicateurs respectifs $CM_1$ , $CM_2$ , ... $CM_n$, le coefficient multiplicateur correspondant à son évolution global est donné par :
\[ CM = CM_1 \times CM_2 \times \cdots \times CM_n \]

}


\exercice{Évolutions successives}{%
\begin{enumerate}
    \item Quelle évolution globale donnent trois augmentations de 5$\%$, 10$\%$ puis 15$\%$ ?
    \item Une quantité augmente de 20$\%$ puis diminue de 30$\%$. Quelle est son évolution globale ?
    \item Une quantité diminue 4 fois de 10$\%$. Quelle est sa diminution globale ?
    \item Une quantité augmente de 25$\%$ puis diminue de 25$\%$. Plus tard elle diminue de 10$\%$ puis augmente de 10$\%$. Quelle est son évolution globale ?
\end{enumerate}
}



\theoreme{Évolution réciproque}{%
Lorsqu'une quantité subit une évolution, elle peut revenir à sa valeur initiale en subissant une \textbf{évolution réciproque}. Si la première évolution correspond à un coefficient multiplicateur $CM$, alors le coefficient multiplicateur réciproque est :
\[ CM_R = \frac{1}{CM} \]
}



\exercice{Évolutions réciproques}{%

	\item Une quantité a diminué de 30$\%$. Quelle augmentation doit-elle subir pour revenir à sa valeur initiale ?
	\item Une quantité augmente deux fois de 10$\%$. Quelle diminution doit-elle subir pour revenir à sa valeur initiale ?
	\item Une quantité augmente de 40$\%$ puis diminue de 25$\%$. Quelle évolution doit-elle subir afin de revenir à sa valeur initiale ?
\end{enumerate}
}

\definition{Fonction affine}{%
On appelle \textbf{fonction affine} toute fonction $f$ définie sur $\RR$ par $\mathbold{f:x\mapsto mx+p}$\\
avec $m$ un réel non nul et $p$ quelconque.
}

\theoreme{Représentation graphique}{%
Soit $f:x\mapsto mx+p$ une fonction affine.\\
Sa \textbf{courbe représentative} est la droite d'équation réduite $y=mx+p$\\
$m$ est le coefficient directeur cette droite, et $p$ est l'ordonnée à l'origine
}

\theoreme{Discriminant}{
À tout polynôme $P:x\mapsto ax^2+bx+c$ on associe le nombre $\Delta=b^2-4ac$ appelé discriminant.

$\bullet$ Si $\Delta<0$ alors $P$ n'admet aucune racine.

$\bullet$ Si $\Delta=0$ alors $P$ admet une unique racine $x_0=\frac{-b}{2a}$

$\bullet$ Si $\Delta>0$ alors $P$ admet deux racines : $x_{1,2}=\frac{-b\pm\sqrt{\Delta}}{2a}$
}

\definition{Partition de l'univers}{

	\phantom{}\qquad $ii)$ $A_i \cap A_j = \emptyset$ \quad $\forall i\neq j$
	
	Moins formellement, il faut que les évènements $(A_1~;~A_2~;~\ldots~;~A_n)$ recouvrent $\Omega$ sans se chevaucher comme représenté sur le schéma :
\end{minipage} \hspace{10mm}

\theoreme{Formule des probabilités totales}{
Soient $(A_1~;~A_2~;~\ldots~;~A_n)$ formant une partition de l'univers. Soit $B$ un évènement quelconque.

On peut calculer la probabilité de l'évènement $B$ par la formule : \\
\phantom{}\hspace{25mm} $ \PP(B) = \PP(A_1 \cap B) + \PP(A_2 \cap B) + \ldots + \PP(A_n \cap B) \left( = \sommek{0}{n} \PP(A_k \cap B)\right) $
}


\remarque{%
Les évènements $A$ et $\barre{A}$ formant toujours une partition de l'univers, on a : $\PP(B)=\PP(A \cap B) + \PP(\barre{A} \cap B)$
}

\subsection{Probabilités conditionnelles}

\definition{Probabilité conditionnelle}{%
\begin{minipage}{0.5\linewidth} \restoreskip \vspace{0pt}
On désigne par $P_A(B)$ la probabilité de l'évènement $B$ sachant de l'évènement $A$ est réalisé.

On le prononce \og probabilité de $B$ sachant $A$ \fg

On peut peut placer ces probabilités conditionnelles sur un arbre pondéré de probabilités comme ci-contre :

$P_A(B)=\dfrac{P(A \cap B)}{P(A)}$
\end{minipage} \hspace{5mm}
\begin{minipage}{0.5\linewidth}
	\begin{tikzpicture}[grow=right, scale = 0.8 ,
	level 1/.style={level distance=4cm, sibling distance=3cm},
	level 2/.style={level distance=3cm, sibling distance=2cm},
	level 3/.style={level distance=2cm, sibling distance=2cm}
	]

		\node {$\Omega$}
		child {node {$\barre{A}$} 
			child {node {$\barre{B}$}
				child {node {$\barre{A}\cap\barre{B}$}
				} edge from parent node[below] {$P_{\barre{A}}(\barre{B})$}
			} 
			child {node {$B$}
				child {node {$\barre{A}\cap B$}
				} edge from parent node[above] {$P_{\barre{A}}(B)$}
			} edge from parent node[below] {$P(\barre{A})$}
		}
		child {node {$A$}
			child {node {$\barre{B}$}
				child {node {$A \cap \barre{B}$}
				} edge from parent node[below] {$P_A(\barre{B})$}
			}
    		child{node {$B$}
    			child {node {$A \cap B$}
				} edge from parent node[above] {$P_A(B)$}
    		} 	edge from parent node[above] {$P(A)$}
    	};
	\end{tikzpicture}
\end{minipage}
}

\subsection{Indépendance}

\definition{Évènements indépendants}{%
On dit que deux évènements $A$ et $B$ sont \textbf{indépendants} si la réalisation de l'un n'a pas d'influence sur la réalisation de l'autre.

Autrement dit si $P_A(B)=P(B)$ \quad et \quad $P_B(A)=P(A)$
}


\theoreme{Indépendance}{%
Deux évènements $A$ et $B$ sont indépendants si et seulement si $P(A \cap B)=P(A) \times P(B)$
}

\subsection{Dérivation et application}

\theoreme{Variations}{
Soit $f$ une fonction définie et dérivable sur $[a;b]$

$f$ est croissante sur $[a;b]$ si et seulement si $f'(x)\geq0 \quad\forall x\in[a;b]$

$f$ est décroissante sur $[a;b]$ si et seulement si $f'(x)\leq0 \quad\forall x\in[a;b]$
}

L'on comprend alors de par cette propriété que pour déterminer les variations d'une fonction il est nécessaire d'étudier le signe de sa fonction dérivée.

Il semble donc nécessaire de faire un rappel quant aux études de signes :

Pour dresser le tableau de signe d'une expression, on la décompose en facteurs simples (Souvent à l'aide d'une factorisation) puis on étudie leurs signes séparément.

\exemple{Résoudre $\dfrac{2x^2-9x+7}{x-3}\geq0$}{
    On commence par étudier le signe du numérateur.\\
    À l'aide du discriminant, on voit que les racines de $2x^2-9x+7$ sont $1$ et $\frac{7}{2}$\\
    Comme ce polynôme a un coefficient dominant positif, il est négatif entre ses racines.\\
    Le signe de $x-3$ s'étudie facilement.\\
    On peut donc remplir le tableau de signes suivant :\\[5pt]
    \begin{tikzpicture}
        \tkzTabInit[lgt=3,espcl=3]{$x$/1, $2x^2-9x+7$/1, $x-3$/1, $\dfrac{2x^2-9x+7}{x-3}$/2}{$-\infty$, $1$, $3$, $\frac{7}{2}$ , $+\infty$}
        \tkzTabLine{,+,z,-,z,-,t,+,}
        \tkzTabLine{,-,t,-,t,+,z,+,}
        \tkzTabLine{,-,z,+,d,-,z,+,}
    \end{tikzpicture}
    
    Ainsi, l'ensemble des solutions de notre inéquation est $S=\left[1~;~3\right[\cup\left]\frac{7}{2};+\infty\right[$

\section{Suites numériques}

\subsection{Modes de définition}

Une suite numérique est une collection infinie de nombres que l'on peut indexer à l'aide des entiers naturels. Il existe deux manières de définir de tels objets mathématiques :

\definition{Génération d'une suite}{
Une suite numérique peut être définie de deux manières : 
\begin{enumerate}
    \item[$\bullet$] \textbf{Génération explicite} : chaque terme est défini directement en fonction de son indice $n$
    \item[$\bullet$] \textbf{Génération par récurrence} : chaque terme est défini à l'aide d'une relation faisant intervenir les termes précédents.
\end{enumerate}
}
\exemple{Génération explicite}{
On pose $u_n=2n+1$\\[5pt]
La suite $\Un$ est définie de manière explicite, et on peut calculer quelques-uns de ses termes :\\[5pt]
$u_0=2\times0+1=1\qquad u_1=2\times1+1=3\qquad u_5=2\times5+1=11\qquad u_{100}=2\times100+1=201$
}
\exemple{Génération par récurrence}{ \restoreskip
On pose $\defsuite{u_0=0}{2u_n+1}$

La suite $\Un$ est définie par récurrence et on peut calculer ses premiers termes :\\[5pt]
$u_1=2\times u_0+1=1\qquad u_2=2\times u_1+1=3\qquad u_3=2\times u_2+1=7 \qquad u_4=2\times u_3+1=15$
}

    \item Soit une suite $(u_n)$ vérifiant la relation de récurrence $u_{n+1}=\frac{1}{1-u_n}$\\

\end{enumerate}

\subsection{Variations et limites}

À l'instar d'une fonction, une suite peut être croissante ou décroissante (Ou bien aucun des deux !) \\
On dira d'une suite $(u_n)$ qu'elle est croissante si chacun de ses termes est supérieur au terme précédent.

\definition{Monotonie}{%
On dit que la suite $(u_n)$ est \textbf{croissante} si $u_{n+1}\geq u_n$ pour tout $\nN$  \\[2pt]
À l'inverse, on dira que $(u_n)$ est \textbf{décroissante} si $u_{n+1}\leq u_n$ pour tout $\nN$

}


\methode{Variations d'une suite}{
Pour déterminer les variations d'une suite $(u_n)$, on calcule la différence $u_{n+1}-u_n$ \\[3pt]
Si $u_{n+1}-u_n \geq 0$, alors la suite $(u_n)$ est croissante.\\
Si $u_{n+1}-u_n \leq 0$, alors la suite $(u_n)$ est décroissante.
}

\definition{Convergence}{%
Si les termes de la suite $(u_n)$ se rapprochent d'une valeur $\ell$ lorsque $n$ devient grand, on dira que $(u_n)$ \textbf{converge} vers $\ell$ \\[2pt]
\qquad On note alors $\limn u_n = \ell$
}



\definition{Limite infinie}{%
Si les termes de la suite $(u_n)$ deviennent de plus en plus grands lorsque $n$ devient grand, on dira que $(u_n)$ \textbf{diverge} vers $+\infty$ \\[2pt]
\qquad On note alors $\limn u_n = +\infty$
}

\section{Fonction exponentielle}

\definition{Fonction exponentielle}{
Il existe une unique fonction $f$ égale à sa propre dérivée et telle que $f(0)=1$

On appelle cette fonction la \textbf{fonction exponentielle} et on note $\mathbold{f(x)= e^x}$
}

$\bullet\, e^a\times e^b=e^{a+b}$ \qquad\quad $\bullet\, \dfrac{e^a}{e^b}=e^{a-b}$ \qquad\quad
$\bullet\, e^{-a}=\dfrac{1}{e^a}$ \qquad\quad $\bullet\, \left(e^a\right)^b=e^{ab}$
}

\subsection{Dérivation et étude de fonctions}

Attention, la fonction exponentielle est égale à sa propre dérivée, mais cette propriété n'est valable que pour \textbf{la} fonction exponentielle ($x\mapsto e^x$).

De manière générale, pour dériver une fonction faisant intervenir des exponentielles, il faut se référer au formulaire de dérivation, et en particulier à la formule suivante :


\methode{Dérivée}{
$\left(e^{u(x)}\right)'=u'(x)\times e^{u(x)}$
}
