\section*{Première partie - Calculs}

\subsection{}
\begin{enumerate}
	\item $\sqrt{17+12\sqrt{2}}=5+\sqrt{8}$
	\item Pour tout nombre réel $x$ différent de $-1$ et $3$ , $\dfrac{1}{x-3}-\dfrac{4}{x^2-2x-3}=\dfrac{1}{x+1}$
	\item Pour tout nombre réel $x$ , $1-2\ln(x^2)=\ln\left(\frac{e}{x^3}\right)-\ln(x)$
	\item Si trois nombres réels non nuls $a$, $b$ et $c$ sont tels que $\frac{1}{a}+\frac{1}{b}+\frac{1}{c}=0$ , alors  \\$(a+b+c)^2=a^2+b^2+c^2$
	\item Si $x$ et $y$ sont deux nombres réels non nuls opposés, alors l'inverse de la moyenne des inverses des carrés de $x$ et $y$ vaut $x^2$
\end{enumerate}



\section*{Deuxième partie - Fonctions}

\subsection{}
\begin{enumerate}
	\item Si la fonction $f$ est convexe sur $\RR$ alors la fonction $g$ définie par $g(x)=e^{f(x)}$ est convexe sur $\RR$
	\item Soit $f$ la fonction définie sur $\RR$ par $f(x) = x^3-x^2-x+1$. La courbe représentative de $f$ admet exactement deux points d'inflexion.
	\item Soient $f$ une fonction définie et deux fois dérivable sur un intervalle $[a~;~b]$ et $\Ccal_f$ sa courbe représentative dans un repère orthonormé. On note $A$ le point de $\Ccal_f$ d'abscisse $a$ et $B$ le point de $\Ccal_f$ d'abscisse $b$. Si $f'$ est décroissante sur $[a~;~b]$ toute tangente à $\Ccal_f$ se situe en-dessous du segment $[AB]$
\end{enumerate}



\subsection{}
Soit $f$ la fonction définie sur $\RR$ par $f(x)=x^2+x+e^{-x}$
\begin{enumerate}
	\item La courbe représentative de la fonction $f$ se situe au-dessus de la courbe représentative de la fonction $f''$ sur l'intervalle $[-2~;~1]$
	\item L'équation $f'(x)=0$ admet une unique solution sur $\RR$
	\item $f$ admet un minimum sur $\RR$ et ce minimum est atteint en $x=0$
\end{enumerate}



\section*{Troisième partie - Suites numériques}

\subsection{}
Soit $(u_n)_{\nN}$ la suite définie par $u_0=1$ et $u_{n+1}=\frac{{u_n}^2}{3}$ pour tout entier naturel $n$. On définit la suite $(v_n)_{\nN}$ par $v_n=\ln(u_n)-\ln(3)$ pour tout entier naturel $n$
\begin{enumerate}
	\item La suite $(v_n)_{\nN}$ est géométrique.
	\item Pour tout $n$ entier naturel, $u_n=3^{1-3^n}$
\end{enumerate}


\subsection{}
\begin{enumerate}
	\item Soit $(u_n)_{\nN}$ une suite divergente. S'il existe une suite $(v_n)_{\nN}$ telle que la suite $(u_n\times v_n)_{\nN}$ converge, alors la suite $(v_n)_{\nN}$ converge.
	\item Soit $(u_n)_{\nN}$ la suite définie par $u_n=\dfrac{(3n+1)^3}{3n^3+1}$ pour tout entier naturel $n$. La suite $(u_n)_{\nN}$ converge vers $3$
	\item Soit $(u_n)_{\nN}$ une suite croissante convergeant vers $5$. $\limn \dfrac{5}{5-u_n}=+\infty$
\end{enumerate}



\section*{Quatrième partie - Probabilités}

\subsection{}
On considère une variable aléatoire $X$ à valeurs dans $\{2~;~4~;~8\}$ et dont l'espérance mathématique vaut $\EE(X)=5$
\begin{enumerate}
	\item $\PP(X=8)\leq \frac{5}{8}$
	\item Si $\PP(X=4)=\frac{1}{2}$ alors $\PP(X=8)=\frac{1}{4}$
	\item $\PP(X=2)+\PP(X=4)\leq \PP(X=8)$
\end{enumerate}



\subsection{}
Soient $A$ et $B$ deux évènements d'un univers $\Omega$
\begin{enumerate}
	\item Les évènements $\barre{B}$ , $A \cap B$ et $\barre{A} \cap B$ forment une partition de $\Omega$
	\item Si $\PP(\barre{A})=\frac{2}{3}$ , $\PP_A(B)=\frac{1}{4}$ et $\PP_{\barre{A}}(B)=\frac{1}{2}$ alors $\PP(\barre{B})=\frac{3}{4}$
	\item Si $\PP(A \cup B)=2\PP(A)$ et $\PP(A)=2\PP(A \cap B)$ alors $3\PP(A)=2\PP(B)$
\end{enumerate}



\section*{Cinquième partie - Géométrie}

\subsection{}
On munit le plan d'un repère orthonormé $\Oij$.

On considère les points $A=(-2~;~1)$ , $B=(2~;~3)$ et $C=(7~;~-2)$
\begin{enumerate}
	\item Le point $H=(3~;~6)$ est le point de concours des hauteurs du triangle $ABC$.
	\item Le point $G=(2~;~-2)$ est le centre du cercle circonscrit au triangle $ABC$.
	\item L'angle $\widehat{ABC}$ est aigu.
\end{enumerate}



\subsection{}
On munit le plan d'un repère orthonormé $\Oij$.
\begin{enumerate}
	\item Soient $a$, $b$, $c$ et $d$ quatre nombres réels quelconques. Les droites $d_1:y=ax+b$ et\\ $d_2:y=cx+d$ sont perpendiculaires si et seulement si $ac=-1$
	\item Soit un segment $[AB]$ de milieu $I$. Pour tout point $M$ du plan , $\scal{AM}{BM}=MI^2-\frac{1}{4}AB^2$
	\item On considère les points $A=(-3~;~-1)$ , $B=(1~;~5)$ et $C=(8~;~1)$. Le triangle $ABC$ est rectangle.
	\item Une équation du cercle de centre $\Omega=(3~;~-1)$ et de rayon $3$ est $x^2+y^2-6x+2y+1=0$
\end{enumerate}
