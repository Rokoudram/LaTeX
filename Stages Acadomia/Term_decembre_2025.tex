\section{Suites et récurrence}

\subsection{Principe de récurrence}
Dans cette section on considèrera $P$ une propriété quelconque dépendant de $n$ un entier naturel.

\definition{Hérédité}{
On dit que la propriété $P$ est \textbf{héréditaire} si $P(n) \Rightarrow P(n+1)$ pour tout $\nN$
}

\theoreme{Principe de récurrence}{
Si une propriété $P$ est vraie au rang $n_0$ et héréditaire, alors elle est vraie pour tout entier $n\geq n_0$
}


\methode{Rédaction d'une récurrence}{\restoreskip \vspace{5pt}
Un raisonnement par récurrence se rédige toujours en 3 étapes : Initialisation, hérédité et conclusion.

Voici une trame de rédaction afin de démontrer une propriété $P$ :

\textbf{\underline{Initialisation} ($n=0$)} :\\[5pt]
\hspace*{10mm}
.................................... donc $P_0$ est vraie

\textbf{\underline{Hérédité}} : \\[5pt]
\hspace*{10mm} On suppose que $P_n$ est vraie pour un certain $\nN$\\
\hspace*{10mm} On montre que $P_{n+1}$ est également vraie. \\
\hspace*{10mm} Par hypothèse de récurrence : .................... \\
\hspace*{57mm} donc .................... \\[5pt]
\hspace*{75mm} $\vdots$ \\
\hspace*{57mm} donc .................... \\
\hspace*{57mm} donc $P_{n+1}$ est vraie \\

\textbf{\underline{Conclusion}} : \\[5pt]
\hspace*{10mm} $P_0$ est vraie et $P$ est héréditaire, donc, par principe de récurrence, $P_n$ est vraie pour tout $\nN$

}


\astuce{
$\bullet$ Pour démontrer une inégalité ou un encadrement, il est souvent nécessaire \og d'élargir \fg\, le résultat obtenu à la fin de l'hérédité

$\bullet$ Si l'objet de la récurrence est une suite vérifiant $u_{n+1}=f(u_n)$ il faut souvent étudier les variations de la fonction $f$ dans un premier temps
}

\exemple{Inégalité de Bernoulli}{\restoreskip \vspace{5pt}
On cherche à démontrer le résultat suivant : $\forall\nN,\forall a>0, \quad (1+a)^n \geq 1+an$ \\
On note $P_n$ : \og $(1+a)^n \geq 1+an$ \fg


\textbf{\underline{Initialisation} ($n=0$)} :\\[5pt]
\hspace*{10mm}
$(1+a)^0=1$ \quad et \quad $1+a\times0=1$ \quad donc $(1+a)^0\geq1+a\times0$ \quad donc $P_0$ est vraie

\textbf{\underline{Hérédité}} : \\[5pt]
\hspace*{10mm} On suppose que $(1+a)^n \geq 1+an$ pour un certain $\nN$

\hspace*{10mm} On montre que $(1+a)^{n+1} \geq 1+a(n+1)$ 

\hspace*{10mm} Par hypothèse de récurrence :

\hspace*{40mm}
$\begin{WithArrows}[format = l]
	\hspace{11.5mm} (1+a)^n \geq 1+an \Arrow{$\times(1+a)$}\\
	\text{donc~~} (1+a)^n \times (1+a) \geq (1+an)\times(1+a) \Arrow{On développe} \\
	\text{donc~~} (1+a)^{n+1} \geq 1+a+an+a^2n  \Arrow{On élargit l'inégalité} \\
	\text{donc~~} (1+a)^{n+1} \geq 1+a(n+1)+a^2n \geq 1+a(n+1) \\
	\text{donc~~} (1+a)^{n+1} \geq1+a(n+1)
\end{WithArrows}$

\textbf{\underline{Conclusion}} : \\[5pt]
\hspace*{10mm} $P_0$ est vraie et $P$ est héréditaire, donc, par principe de récurrence, $P_n$ est vraie pour tout $\nN$
}

}


\exercice{Inégalité simple}{\restoreskip

Démontrer par récurrence que $u_n>n$ pour tout $n$ entier naturel
}


\exercice{Extension de l'inégalité}{\restoreskip

}

	\item Dresser le tableau de variations de la fonction $f$
	\item Montrer par récurrence que $0<u_n<2$ pour tout $\nN$
\end{enumerate}
}



\subsection{Convergence et divergence}

\definition{Suite convergente}{\restoreskip \vspace{5pt}
On dit qu'une suite $\mathbold{(u_n)}$ \textbf{converge vers un réel} $\mathbold\ell$ si : 
\[ \forall \epsilon>0 , \quad \exists n_\epsilon\in\NN , \quad \left( n\geq n_\epsilon \Rightarrow \abs{u_n-\ell}<\epsilon \right) \]

De manière plus intuitive, on dit que $\mathbold{(u_n)}$ converge vers $\mathbold\ell$ si tout intervalle ouvert contenant $\ell$ contient également une infinité de termes consécutifs de la suite $(u_n)$

}

Pour plus de clarté, voici une figure représentant une suite convergente : 

\begin{tikzpicture}
	\begin{axis}[clip=false , width=19.5cm , height=7.5cm ,
    	xmin=0,xmax=39,ymin=2,ymax=6.9, ytick distance=1,xtick distance=5, yticklabels=\empty ,
        axis lines=middle, axis line style = thick, grid=major, ticklabel style = {font=\small} ,
        ]
        
		\addplot[domain=0:39, samples at ={1,...,39} , only marks , mark = * , mark size = 3pt , red]
		{4 + (6-6*sin(deg(\x)))/(x+1)};
		
		\draw[dashed , line width = 2pt , color = blue] (0,4) -- (39,4) ;
		\draw[fill=blue,opacity=0.15] (0,4.6) rectangle (39,3.4) ;
		\draw[color=blue] (0,4.6) node[left] {$\ell+\epsilon$} ;
		\draw[color=blue] (-0.5,4) node[left] {$\ell$} ;
		\draw[color=blue] (0,3.4) node[left] {$\ell-\epsilon$} ;
		\draw (-1,6.5) node{$u_n$} ;
		\draw (38,1.7) node {$n$} ;
		\draw[dashed , color=red , line width = 1.5pt] (19,4.2) -- (19,2) node[below] {$n_\epsilon$} ;
	\end{axis}
\end{tikzpicture}

On voit que tous les termes de la suite $(u_n)$ sont compris dans l'intervalle $[\ell-\epsilon;\ell+\epsilon]$ à partir du rang $n_\epsilon = 19$. Pour une plus petite valeur de $\epsilon$, le rang $n_\epsilon$ aurait été plus élevé.



\definition{Divergence}{
On dit d'une suite $(u_n)$ qu'elle \textbf{diverge} si elle n'est pas convergente.
}


\theoreme{Théorème de l'encadrement}{\restoreskip \vspace{5pt}
Si $(u_n)$ , $(v_n)$ et $(w_n)$ sont trois suites telles que l'on ait : 

\hspace{25mm}  $u_n \leq v_n \leq w_n \quad \forall\nN$ \qquad et \qquad $\limn u_n = \limn w_n = \ell \in \RR$

Alors la suite $(v_n)$ converge et $\limn v_n =\ell$
}


\theoreme{Comparaison asymptotique}{\restoreskip \vspace{5pt}
Si $(u_n)$ et $(v_n)$  sont deux suites telles que l'on ait : 

\hspace{25mm}  $u_n \leq v_n  \quad \forall\nN$ \qquad et \qquad $\limn u_n = +\infty$

Alors la suite $(v_n)$ diverge et $\limn v_n =+\infty$
}

\definition{Convexité}{
Soit $f$ une fonction réelle\\[5pt]
$\bullet\,$ $f$ est convexe $\Longleftrightarrow$ $f'$ est croissante $\Longleftrightarrow$ $f''$ est positive\\
$\bullet\,$ $f$ est concave $\Longleftrightarrow$ $f'$ est décroissante $\Longleftrightarrow$ $f''$ est négative
}

\theoreme{Courbe convexe}{
$\bullet$ Si $f$ est convexe sur un intervalle $I$, alors $\Ccal_f$ se situe en-dessous de ses tangentes et au-dessus de ses cordes sur l'intervalle $I$\\[5pt]
$\bullet$ Si $f$ est concave sur un intervalle $I$, alors $\Ccal_f$ se situe au-dessus de ses tangentes et en-dessous de ses cordes sur l'intervalle $I$ 
}
