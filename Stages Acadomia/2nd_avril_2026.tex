\subsection{Fractions}

\theoreme{Propriété des fractions}{
Soient $a$ , $b$ , $c$ et $d$ quatre nombres réels non nuls. \\

$\bullet\, \dfrac{a}{d} + \dfrac{b}{d}=\dfrac{a+b}{d}$ \qquad $\bullet\, \dfrac{a}{b} + \dfrac{c}{d} = \dfrac{ad+bc}{bd}$ \qquad $\bullet\, \dfrac{a}{b}\times\dfrac{b}{d}=\dfrac{ab}{cd}$ \qquad $\bullet\, \dfrac{\,\frac{a}{b}\,}{\frac{b}{d}} = \dfrac{ad}{bc}$
}

\methode{Calcul de fractions}{
Pour additionner ou soustraire deux fractions de dénominateur différent il faut les transformer en fractions équivalentes de dénominateur commun
}

\exemple{Calculs de fractions}{
$\bullet\, \frac{8}{7}+\frac{3}{7}=\frac{8+3}{7}=\frac{11}{7}$ \qquad\qquad $\bullet\, \frac{2}{3}\times\frac{12}{5}=\frac{2\times12}{3\times5}=\frac{24}{15}=\frac{3\times8}{3\times5}=\frac{8}{5}$ \\[5pt]

$\bullet\, \frac{7}{12}-\frac{1}{3}=\frac{7}{12}-\frac{1\times4}{3\times4}=\frac{7}{12}-\frac{4}{12}=\frac{7-4}{12}=\frac{3}{12}=\frac{3\times1}{3\times4}=\frac{1}{4}$
}

\subsection{Puissances}

\theoreme{Propriété de calcul}{
Soient $x$ et $y$ deux nombres réels. Soient $a$ et $b$ deux nombres entiers. \\[5pt]
$\bullet\, x^a\times x^b=x^{a+b}$ \qquad $\bullet\, \dfrac{x^a}{x^b}=x^{a-b}$ \qquad $\bullet\, \left(x^a\right)^b=x^{ab}$ \qquad $\bullet\, x^a\times y^a=(xy)^a$ \qquad $\bullet\, \dfrac{x^a}{y^a}=\left(\dfrac{x}{y}\right)^a$
}


\exemple{Simplification de puissances}{
 $\dfrac{3^8\times9^3}{3^5\times27^2}=\dfrac{3^8\times{(3^2)}^3}{3^5\times{(3^3)}^2}=\dfrac{3^8\times3^{2\times3}}{3^5\times3^{3\times2}} = \dfrac{3^8\times3^6}{3^5\times3^6}=\dfrac{3^{8+6}}{3^{5+6}}=\dfrac{3^{14}}{3^{11}}=3^{14-11}=3^3=27$
}

\subsection{Sur les vecteurs}

\definition{Vecteur}{
\begin{minipage}{0.6\linewidth}
    Un \textbf{vecteur} est un objet mathématique permettant de décrire un mouvement de \textbf{translation} que l'on représente par une flèche. \\[5pt]    
    Il est caractérisé par 3 paramètres : \\[5pt]    
    $\bullet$ Sa \textbf{direction} $\rightarrow$ la droite sur laquelle il s'aligne \\
    $\bullet$ Son \textbf{sens} $\rightarrow$ le côté choisi dans sa direction \\
    $\bullet$ Sa \textbf{norme} $\rightarrow$ la longueur du déplacement \\[5pt]
    On note les vecteurs avec des flèches au-dessus de leur nom.
\end{minipage}
\begin{minipage}{0.35\linewidth}
    \begin{tikzpicture}[scale=0.8]
        \draw[line width=2pt,-Stealth] (-1,-2) node {\Large$\bullet$} -- node[below right] {$\mathbold{\vv{u}}$} (3,1);
        \draw[dashed] (5,2.5)--(-2,-2.75);
        \draw [thick,-Stealth,color=blue] (3.5,-0.5)  node[right] {Sens} to [out=180,in=-90] (2.8,0.5);
        \draw [thick,-Stealth,color=blue] (2.5,2.5)  node[left] {Direction} to [out=15,in=120] (4,2);
        \draw [thick,color=blue, Stealth-Stealth] (-1.25,-1.7)-- node[above,rotate=37] {Norme} (2.7,1.25) ;
    \end{tikzpicture}
\end{minipage}
}


\definition{Entre deux points}{
Soient $A$ et $B$ deux points distincts du plan.\\
Pour décrire la translation permettant de relier le point $A$ au point $B$ on parlera du vecteur $\mathbold{\vv{AB}}$
}


\theoreme{Caractéristiques du vecteur $\vv{AB}$}{ \vspace{3pt}
\begin{enumerate}
    \item La direction du vecteur $\vv{AB}$ est la même que celle de la droite $(AB)$
    \item Le sens de $\vv{AB}$ est \og de $A$ vers $B$ \fg
    \item La norme de $\vv{AB}$ est la longueur $AB$
\end{enumerate}
}



\definition{Vecteur nul}{
Pour décrire l'absence de déplacement, on utilise le \textbf{vecteur nul} noté $\vv{0}$
}


\theoreme{Milieu d'un segment}{
\begin{minipage}{0.4\linewidth}
	Le point $M$ est le \textbf{milieu} du segment $[AB]$ si et seulement si $\vv{AM}=\vv{MB}$
\end{minipage}
\begin{minipage}{0.5\linewidth}
	\begin{tikzpicture}[scale=0.6]
    	\draw[-Stealth , line width = 1pt] (0,0) node{x} node[below] {$A$} -- (5.825,0.975)  ;
    	\draw[-Stealth , line width = 1pt] (6,1) node{x} node[below] {$M$} -- (11.825,1.975) ;
    	\draw (12,2) node{x} node[below] {$B$} ;
	\end{tikzpicture}
\end{minipage}
}



\exercice{Milieu}{\vspace{5pt}
\begin{enumerate}
	\item Construire un triangle $ABC$ quelconque.
	\item Placer le point $D$ tel que $\vv{CD}=\vv{AB}$
	\item Placer le point $E$ tel que $\vv{EC}=\vv{AB}$
	\item Que peut-on dire du point $C$ par rapport au segment $[DE]$ ? Justifier.
\end{enumerate}
}



\theoreme{Parallélogramme}{
$ABCD$ est un \textbf{parallélogramme} si et seulement si $\vv{AB}=\vv{DC}$
}



\exercice{Extension d'un triangle}{

Quelle est la nature du quadrilatère $ABMC$ ?
}

\begin{minipage}{0.6\linewidth}
    Soient deux vecteurs $\vv{u}$ et $\vv{v}$.
    
    Si l'on enchaîne successivement les deux translations décrites par les vecteurs $\vv{u}$ et $\vv{v}$ alors on dira que l'on a effectué une translation de vecteur $\vv{u}+\vv{v}$
\end{minipage}

}

On voit sur la figure ci-dessus qu'il est essentiel d'effectuer la translation de vecteur $\vv{v}$ en partant du point d'arrivée de première translation.
}




\begin{minipage}{0.55\linewidth}
    \definition{Vecteurs opposés}{\restoreskip \vspace{5pt}
    On dit que deux vecteurs $\vv{u}$ et $\vv{v}$ sont \textbf{opposés} si \vspace{-10pt} $$\vv{u}+\vv{v}=\vv{0}$$\\[-20pt]
    On note alors $\vv{v}=-\vv{u}$
    }
\end{minipage}
\begin{minipage}{0.4\linewidth}
    \theoreme{Vecteurs opposés}{
    Pour tous points $A$ et $B$, $\vv{AB}+\vv{BA}=\vv{0}$
    }
\end{minipage}

\end{minipage} \hspace{5mm}

}

\theoreme{Relation de Chasles}{

}

\begin{enumerate} \setlength{\itemsep}{1pt}

\end{enumerate}
}



\definition{Colinéarité}{
On dit de deux vecteurs $\vv{u}$ et $\vv{v}$ qu'ils sont colinéaires s'ils ont la même direction.
}


\theoreme{Colinéarité}{
Deux vecteurs $\vv{u}$ et $\vv{v}$ sont colinéaires si et seulement si il existe $k\in\RR$ tel que $\vv{u}=k\times\vv{v}$
}

\remarque{Cette propriété vient du fait que lorsque l'on multiplie un vecteur par un nombre, sa norme peut être multipliée, son sens peut changer, mais sa direction est toujours identique.}


\exercice{Relation de colinéarité}{\restoreskip
On considère quatre points distincts $A$, $B$, $C$ et $D$.\\
Dans chacune des situations suivantes, montrer que les vecteurs $\vv{u}$ et $\vv{v}$ sont colinéaires :

a) $\vv{u}=2\vv{AB}-\vv{AC}$ \qquad et \qquad $\vv{v}=6\vv{AB}-3\vv{AC}$

b) $\vv{u}=3\vv{BC}+\vv{AD}$ \qquad et \qquad $\vv{v}=6\vv{BC}-2\vv{DA}$

c) $\vv{u}=4\vv{AB}-2\vv{BD}$ \qquad et \qquad $\vv{v}=3\vv{BD}+6\vv{BA}$

d) $\vv{u}=2\vv{AB}+\vv{BC}$ \qquad et \qquad $\vv{v}=2\vv{AB}+2\vv{AC}$
}



\begin{minipage}{0.48\linewidth}
    \theoreme{Alignement}{
    Trois points du plan $A$, $B$ et $C$ sont alignés si et seulement si $\vv{AB}$ et $\vv{AC}$ sont colinéaires.
    }
\end{minipage}\hfill
\begin{minipage}{0.48\linewidth}
    \remarque{
    On aurait pu choisir les vecteurs $\vv{BC}$ et $\vv{BA}$, ou bien les vecteurs $\vv{CA}$ et $\vv{CB}$
    }
\end{minipage}

\subsection{Systèmes de coordonnées}

Dans cette section le plan sera rapporté au repère orthonormé $(O~;~I;~;J)$

\theoreme{Milieu}{
On considère les points $A\smeq(x_A~;~y_A)$ et $B\smeq(x_B~;~y_B)$ et on note $\mathbold M$ le \textbf{milieu du segment} $[AB]$\\
Les coordonnées du point $M$ sont alors $\left(\frac{x_A+x_B}{2}~;~\frac{y_A+y_B}{2}\right)$
}

Soient les points $A\smeq(x_A~;~x_B)$ et $B\smeq(x_B~;~y_B)$

\theoreme{Calcul vectoriel}{
Lorsque l'on somme ou que l'on multiplie des vecteurs par des réels, il suffit d'effectuer les \textbf{opérations} sur chacune de leurs \textbf{coordonnées}.
}

\exemple{Somme et produit de vecteurs}{
Soient $\vcoord{u}{-1\\3}$ et $\vcoord{v}{2\\2}$. On pose $\vv{w}=2\vv{v}-3\vv{u}$\\
On a alors $\vv{w}=2\times\matrice{2\\2}-3\times\matrice{-1\\3}=\matrice{4\\4}+\matrice{3\\-9}=\matrice{7\\-5}$
}

Alors le vecteur $\vv{AB}$ a pour coordonnées $\vcoord{AB}{x_B-x_A\\y_B-y_A}$
}

\exemple{Déterminer un domaine de définition}{
On considère la fonction $f:x\mapsto\dfrac{x+4}{x-1}$

\subsection{Variations et extremums}

\definition{Notion de croissance}{
On considère un intervalle $I$

$\bullet$ On dit qu'une fonction $f$ est \textbf{croissante} sur $I$ si pour tous réels $a,b\in I$ , $a\leq b \Longleftrightarrow f(a)\leq f(b)$

\hspace{1cm} Intuitivement on dira que $f$ conserve les inégalités.

$\bullet$ On dit qu'une fonction $f$ est \textbf{décroissante} sur $I$ si pour tous réels $a,b\in I$ , $a\leq b \Longleftrightarrow f(a)\geq f(b)$

\hspace{1cm} Intuitivement on dira que $f$ inverse les inégalités.
}



\exercice{Monotonie parabolique}{
On considère la fonction $g$ définie sur $\RR$ par $g(x)=x^2-2x$

Soient $a$ et $b$ deux réels quelconques.
\begin{enumerate}
	\item Montrer que $f(b)-f(a)=(b-a)(a+b-2)$
	\item On suppose que $1<a<b$
	
	\begin{enumerate}
		\item Montrer que $f(b)-f(a)>0$
		\item En déduire que la fonction $f$ est croissante sur l'intervalle $[1;+\infty[$
	\end{enumerate}
\end{enumerate}
}



\begin{minipage}{0.6\linewidth}
\definition{Extremums}{
On considère un intervalle $I$

$\bullet$ On dit que $M$ est le \textbf{maximum} de la fonction $f$ sur $I$ si :

\hspace{1cm} $i)$ Pour tout $x\in I$ , $f(x) \leq M$ \\[1pt]
\phantom{} \hspace{8.5mm} $ii)$ Il existe un certain $x_0\in I$ tel que $f(x_0)=M$

On dira alors que la fonction $f$ atteint son maximum en $x_0$

$\bullet$ On dit que $m$ est le \textbf{minimum} de la fonction $f$ sur $I$ si :

\hspace{1cm} $i)$ Pour tout $x\in I$ , $f(x) \geq m$ \\[1pt]
\phantom{} \hspace{8.5mm} $ii)$ Il existe un certain $x_0\in I$ tel que $f(x_0)=m$

On dira alors que la fonction $f$ atteint son minimum en $x_0$
}
\end{minipage} \hspace{5mm}
\begin{minipage}{0.3\linewidth}
	\definition{Monotonie}{
	On dit qu'un fonction $f$ est \textbf{monotone} sur un intervalle $I$ si elle y est toujours croissante, ou toujours décroissante.
	}
\end{minipage}

\definition{Variation absolue}{
Si une quantité ayant une valeur initiale $V_I$ évolue et atteint une valeur finale $V_F$, on dira que sa \textbf{variation absolue} est la quantité $\Delta V=V_F-V_I$
}

\definition{Taux d'évolution}{\restoreskip
Le \textbf{taux d'évolution} est un nombre positif permettant de décrire la \textbf{variation relative }d'une quantité.

Si une quantité ayant une valeur initiale $V_I$ évolue et atteint une valeur finale $V_F$, on dira que sa \textbf{variation relative} est la quantité $t=\dfrac{V_F-V_I}{V_I}$

Une hausse de $x\%$ correspond à un taux d'évolution $t=\frac{x}{100}$\\
Une diminution de $x\%$ correspond à un taux d'évolution $t=\frac{-x}{100}$
}



\exercice{Variations}{
Dans chacune des situations suivantes, déterminer la variations absolue et la variation relative :
\begin{enumerate}
    \item Une veste coûtant 250\euro\, est soldée à 175\euro
    \item La commande d'Ava coûtait 50\euro, mais en incluant les frais de port et les taxes elle devra finalement payer 65\euro
    \item Raj a préparé 6L de soupe. Après avoir servi tout le monde, il n'en reste plus qu'un quart.

\end{enumerate}
}

\exemple{Déterminer une valeur initiale}{\restoreskip
Un article dans un magasin est soldé à -40\% et coûte actuellement 72\euro. Quel était son prix de vente initial?

Une diminution du prix de 40\% correspond à un taux d'évolution $t=-0,4$\\
De plus la valeur finale de l'article est $V_F=72$\\
En reprenant la formule ci-dessus, on obtient alors :

$-0,4=\dfrac{72-V_I}{V_I} \Leftrightarrow -0,4\times V_I=72-V_I \Leftrightarrow V_I-0,4V_I=72 \Leftrightarrow 0,6V_I=72 \Leftrightarrow V_I=\dfrac{72}{0,6}=120$
}

\definition{Coefficient multiplicateur}{
Si une quantité ayant une valeur initiale $V_I$ évolue et atteint une valeur finale $V_F$, on dira que le \textbf{coefficient multiplicateur} associé à cette évolution est $CM=\dfrac{V_F}{V_I}$
}


\theoreme{$CM$ et taux}{
Le coefficient multiplicateur et le taux d'évolution sont liés par une formule simple :
$CM=1+t$
}

\theoreme{Évolutions successives}{
Si une quantité subit des évolutions successives de coefficients multiplicateurs respectifs $CM_1$, $CM_2$, \ldots et $CM_n$, alors elle subit une évolution globale dont le coefficient multiplicateur est :
\[CM_G=CM_1\times CM_2\times\ldots CM_n\]
}

\section{Résolution d'équations}

\exemple{Équation du premier degré}{
    $\begin{WithArrows}[format = cc]
    & 3x-2=8-7x   \Arrow{On ajoute 2 de chaque côté} \\
    \Leftrightarrow  & 3x=10-7x \Arrow{On ajoute $7x$ de chaque côté} \\
    \Leftrightarrow  & 10x=10\Arrow{On divise par 10 de chaque côté} \\
    \Leftrightarrow  & x=1 
    \end{WithArrows}$
}

\exemple{Équation produit nul}{
    $\begin{WithArrows}[format = lccc]
    & & (3x-2)(8-7x)=0 & \Arrow{On sépare les facteurs} \\
    \Leftrightarrow\; & 3x-2=0 & \text{ou} & 8-7x=0 \Arrow[jump=2]{On résout séparément} \\
    \Leftrightarrow & 3x=2 & \text{ou} & 7x=8 \\
    \Leftrightarrow & x=\frac{2}{3} & \text{ou} & x=\frac{8}{7}
    \end{WithArrows}$ \\[5pt]
    On obtient alors $S=\left\{\dfrac{2}{3}~~;~~\dfrac{8}{7}\right\}$
}

\methode{Mise en équation}{
Pour résoudre un problème à l'aide d'une équation on peut distinguer quelques étapes essentielles :
\begin{enumerate}
    \item Repérer les valeurs qui nous sont inconnues
    \item En choisir une qui sera notre inconnue principale : $x$
    \item Utiliser l'énoncé pour exprimer les autres valeurs à l'aide de $x$
    \item Repérer des relations entre ces valeurs pour écrire une équation
    \item Résoudre l'équation
\end{enumerate}
}
