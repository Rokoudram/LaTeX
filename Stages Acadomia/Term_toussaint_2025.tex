Pour faire disparaître des racines carrées embêtantes (par exemple au dénominateur d'une fraction), on peut utiliser l'identité remarquable : $(a-b)(a+b)=a^2-b^2$

\subsection{Variations}

\theoreme{Variations}{
Soit $f$ une fonction définie et dérivable sur $[a;b]$\\
Si $f'(x)\geq0 \quad\forall x\in[a;b]$ alors $f$ est croissante sur $[a;b]$\\
Si $f'(x)\leq0 \quad\forall x\in[a;b]$ alors $f$ est décroissante sur $[a;b]$
}


L'on comprend alors de par cette propriété que pour déterminer les variations d'une fonction il est nécessaire d'étudier le signe de sa fonction dérivée.

Il semble donc nécessaire de faire un rappel quant aux études de signes :

Pour dresser le tableau de signe d'une expression, on la décompose en facteurs simples (Souvent à l'aide d'une factorisation) puis on étudie leurs signes séparément.



\exemple{Dresser le tableau de variations de $f:x\mapsto x(3-\sqrt{x})$ }{ \restoreskip
	On commence par noter que $f$ est définie sur $\RR_+$
	
	On remarque ensuite que $f(x)$ est de la forme $u(x)v(x)$ \qquad avec $\begin{cases}u(x)=x\\v(x)=3-\sqrt{x}\end{cases}$
	
	Ainsi : $f'(x)=u'(x)v(x)+u(x)v'(x)$ \qquad avec $\begin{cases}u'(x)=1\\v'(x)=-\frac{1}{2\sqrt{x}}\end{cases}$
	
	On obtient alors :
	
    $\begin{WithArrows}[format = ll]
	f'(x) & = 1\times(3-\sqrt{x})+x\times\left(-\frac{1}{2\sqrt{x}}\right) = 3-\sqrt{x} - \frac{x}{2\sqrt{x}} \Arrow{On utilise : $x=\sqrt{x}\times\sqrt{x}$} \\
	& = 3- \sqrt{x} - \frac{\sqrt{x}\times\sqrt{x}}{2 \sqrt{x}} \Arrow{On simplifie et on met au même dénominateur} \\
	&= 3- \frac{2\sqrt{x}}{2}-\frac{\sqrt{x}}{2} \\
	& =3-\frac{3\sqrt{x}}{2} 
    \end{WithArrows}$
    
    On s'intéresse maintenant au signe de $f'(x)$ :
    
        $\begin{WithArrows}[format = ll]
	f'(x) > 0 & \Leftrightarrow 3-\frac{3\sqrt{x}}{2} > 0 \\
	& \Leftrightarrow 3 > \frac{3\sqrt{x}}{2}  \Arrow{$\times\frac{2}{3}$} \\
	& \Leftrightarrow 2 > \sqrt{x} \Arrow{On applique la fonction carré} \\
	& \Leftrightarrow 4 > x
    \end{WithArrows}$
    
    On obtient alors le tableau de variations suivant : 

	\begin{tikzpicture}
		\tkzTab[lgt=3,espcl=5]{ $x$ / 1 , $f'(x)$ / 1 , Variations \\ de $f$ / 2}
		{ $0$ , $4$ ,$+\infty$}
		{ ,+,z,-,}
		{ -/$0$ , +/$4$ , -/}%
	\end{tikzpicture}
}

\definition{Convexité}{
Soit $f$ une fonction réelle \\

$\bullet\,$ $f$ est convexe $\Leftrightarrow$ $f'$ est croissante $\Leftrightarrow$ $f''$ est positive\\

$\bullet\,$ $f$ est concave $\Leftrightarrow$ $f'$ est décroissante $\Leftrightarrow$ $f''$ est négative
}

\section{Fonction exponentielle}

\definition{Fonction exponentielle}{
Il existe une unique fonction $f$ égale à sa propre dérivée et telle que $f(0)=1$

On appelle cette fonction la \textbf{fonction exponentielle} et on note $\mathbold{f(x)= e^x}$
}

De cette définition découlent de nombreuses propriétés que nous n'allons pas démontrer lors de ce stage. Elles sont en revanche essentielles à connaître et à savoir utiliser.

\begin{center}
	\begin{minipage}{0.4\linewidth}
		\theoreme{Propriétés de calcul}{
		\begin{center}
		    $\bullet\, e^a\times e^b=e^{a+b}$ \qquad\quad $\bullet\, \dfrac{e^a}{e^b}=e^{a-b}$\\[8pt]
		    $\bullet\, e^{-a}=\dfrac{1}{e^a}$ \qquad\quad $\bullet\, \left(e^a\right)^b=e^{ab}$
		\end{center}
		}
	\end{minipage}\hspace{5mm}
	\begin{minipage}{0.5\linewidth}
		\theoreme{Propriétés de la fonction}{
		\begin{center}
		   $\bullet$ Pour tout $x$ réel, $e^x>0$\\[2pt]   
		   $\bullet$ La fonction exponentielle est strictement croissante.\\[2pt]
		   $\bullet\, e^a=e^b\Leftrightarrow a=b$\\[2pt]
		   $\bullet\, e^a<e^b\Leftrightarrow a<b$
		\end{center}
		}
	\end{minipage}
\end{center}


\exemple{Simplification}{
$A= \dfrac{\left(e^{x+1}\times e^{-2}\right)^2}{e^{2x-3}}=\dfrac{\left(e^{x+1-2}\right)^2}{e^{2x-3}} = \dfrac{\left(e^{x-1}\right)^2}{e^{2x-3}} = \dfrac{e^{2x-2}}{e^{2x-3}} = e^{2x-2-(2x-3)} = e^1 =e $
}

\subsection{Fonctions trigonométriques}

\definition{Périodicité}{
On dit qu'une fonction $f$ est \textbf{périodique} de période $p$ si pour tout $x\in\Dcal_f$  : $f(x+p)=f(x)$ 
}

\definition{Parité}{
On dit qu'une fonction $f$ est \textbf{paire} si pour tout  $x\in\Dcal_f$ , $\mathbold{f(-x)=f(x)}$ \\[5pt]
On dit qu'une fonction $f$ est \textbf{impaire} si pour tout  $x\in\Dcal_f$ , $\mathbold{f(-x)=-f(x)}$
}

\theoreme{Propriétés des fonctions trigonométriques}{
Les fonctions $\cos$ et $\sin$ sont toutes les deux $2\pi$-périodiques. \\[5pt]
La fonction $\cos$ est paire et la fonction $\sin$ est impaire.
}
