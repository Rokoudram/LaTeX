\section{Calcul numérique}

\subsection{Fractions}

\theoreme{Propriété des fractions}{
Soient $a$ , $b$ , $c$ et $d$ quatre nombres réels non nuls. \\

$\bullet\, \dfrac{a}{d} + \dfrac{b}{d}=\dfrac{a+b}{d}$ \qquad $\bullet\, \dfrac{a}{b} + \dfrac{c}{d} = \dfrac{ad+bc}{bd}$ \qquad $\bullet\, \dfrac{a}{b}\times\dfrac{b}{d}=\dfrac{ab}{cd}$ \qquad $\bullet\, \dfrac{\,\frac{a}{b}\,}{\frac{b}{d}} = \dfrac{ad}{bc}$
}

\methode{Calcul de fractions}{
Pour additionner ou soustraire deux fractions de dénominateur différent il faut les transformer en fractions équivalentes de dénominateur commun
}

\exemple{Calculs de fractions}{
$\bullet\, \frac{8}{7}+\frac{3}{7}=\frac{8+3}{7}=\frac{11}{7}$ \qquad\qquad $\bullet\, \frac{2}{3}\times\frac{12}{5}=\frac{2\times12}{3\times5}=\frac{24}{15}=\frac{3\times8}{3\times5}=\frac{8}{5}$ \\[5pt]

$\bullet\, \frac{7}{12}-\frac{1}{3}=\frac{7}{12}-\frac{1\times4}{3\times4}=\frac{7}{12}-\frac{4}{12}=\frac{7-4}{12}=\frac{3}{12}=\frac{3\times1}{3\times4}=\frac{1}{4}$
}

\subsection{Puissances}

\theoreme{Propriété des puissances}{
Soient $x$ un nombre réel et $n$ un nombre entier. \\[5pt]
$\bullet\, x^0=1$ \qquad $\bullet\, x^{-n}=\dfrac{1}{x^n}$ \qquad $\bullet\, x^n=\begin{cases}\abs{x}^n \text{ si } n \text{ est pair}\\ -\abs{x}^n \text{ si } n \text{ est impair} \end{cases}$
}



\theoreme{Propriété de calcul}{
Soient $x$ et $y$ deux nombres réels. Soient $a$ et $b$ deux nombres entiers. \\[5pt]
$\bullet\, x^a\times x^b=x^{a+b}$ \qquad $\bullet\, \dfrac{x^a}{x^b}=x^{a-b}$ \qquad $\bullet\, \left(x^a\right)^b=x^{ab}$ \qquad $\bullet\, x^a\times y^a=(xy)^a$ \qquad $\bullet\, \dfrac{x^a}{y^a}=\left(\dfrac{x}{y}\right)^a$
}

\section{Polynômes du second degré}


\begin{minipage}{0.55\linewidth}
\definition{Différentes formes}{
On considère un polynôme du second degré $P$\\[5pt]
La forme développée de $P$ est $P(x)=ax^2+bx+x$\\[5pt]
La forme canonique de $P$ est $P(x)=a(x-\alpha)^2+\beta$\\[5pt]
La forme factorisé de $P$ est $P(x)=a(x-x_1)(x-x_2)$
}
\end{minipage}\hfill
\begin{minipage}{0.4\linewidth}
    \begin{tikzpicture}
        \begin{axis}[xmin=-2,xmax=6,ymin=-4,ymax=4,axis lines=middle,ytick distance=1,xtick distance=1,
        grid=major, ticklabel style = {opacity=0}, enlargelimits = { abs = 0.2 }]
            \addplot[color=red,line width=1.5pt,samples=100,domain=-5:7] {0.5*(x+1)*(x-4)};
            \draw[color=blue] (-1.5,1/3) node {$\mathbf{x_1}$};
            \draw[color=blue] (4.5,1/3) node {$\mathbf{x_2}$};
            \draw[color=red] (3.5,-2.5) node {$\mathbf{\Ccal}$};
            \draw[color=blue,dashed,thick] (3/2,0) node[above] {$\alpha$} -- (3/2,-25/8) -- (0,-25/8) node[left] {$\beta$};
        \end{axis}
    \end{tikzpicture}
\end{minipage}

\definition{Parabole}{
La courbe représentative du polynôme $P$ est toujours une parabole
\begin{enumerate}
\item[$\bullet$] L'orientation de la parabole dépend du signe de $a$
\item[$\bullet$] La parabole du polynôme $P$ a pour sommet le point $S(\alpha,\beta)$
\item[$\bullet$] $\alpha=\frac{-b}{2a}$ et $\beta=f(\alpha)$
\item[$\bullet$]  Les points d'intersection entre l'axe des abscisse et $\Ccal$ correspondent aux racines de $P$

\end{enumerate}
}

\theoreme{Discriminant}{
Au polynôme $P$ on associe le nombre $\Delta=b^2-4ac$ appelé discriminant.

\begin{enumerate}
    \item[$\bullet$] Si $\Delta<0$ alors $P$ n'admet aucune racine.
    \item[$\bullet$] Si $\Delta=0$ alors $P$ admet une unique racine $x_0=\frac{-b}{2a}$
    \item[$\bullet$] Si $\Delta>0$ alors $P$ admet deux racines : $x_{1,2}=\frac{-b\pm\sqrt{\Delta}}{2a}$
    \end{enumerate}
}

Quand il atteint le train $B$, il se retourne à nouveau et fonce vers le train $A$, et ainsi de suite.\\

Quand les deux trains se rencontreront, quelle distance le faucon pélerin aura-t-il parcouru ?
